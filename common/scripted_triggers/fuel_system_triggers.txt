fuel_system_is_member_of_prev = {
	NOT = { tag = PREV }
	NOT = { is_in_array = { GLOBAL.fuel_system_manager = THIS.id } }
	OR = {
		is_in_faction_with = PREV
		AND = {
			PREV = { tag = GER }
			OR = { tag = FIN tag = ITA tag = HUN tag = ROM tag = BUL tag = CZE tag = SPR tag = VIC }
		}
		AND = {
			PREV = { tag = ENG }
			OR = { tag = POL tag = FRA tag = CAN tag = AST tag = NZL tag = SAF tag = RAJ }
		}
		AND = {
			PREV = { tag = JAP }
			OR = { tag = MAN tag = SIA }
		}
	}
}
