HOL_build_less_in_mainland = {
    allowed = { 
        original_tag = HOL 
    }
    enable = {
        always = yes
    }
    abort = {
        date > 1940.01.01
    }
    
    # 1. Heavily incentivize state 309 (Batavia)
    ai_strategy = {
        type = factory_build_score_factor
        target = 7
        value = 500  # Massive modifier to force Batavia to the top of the AI's queue
    }

    # 2. Block the mainland states by forcing their score to 0
    # Repeat this exact block for HOL's mainland state IDs (e.g., Holland, Friesland, Brabant)
    ai_strategy = {
        type = factory_build_score_factor
        target = 36  # State ID for Holland
        value = -100  # Negative value zero-scores the state's viability
    }
    ai_strategy = {
        type = factory_build_score_factor
        target = 35  # State ID for Friesland
        value = -100
    }
}
