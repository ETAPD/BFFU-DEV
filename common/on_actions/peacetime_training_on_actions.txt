on_actions = {

	# Grant the Peacetime Training QoL spirit to every country at game start.
	on_startup = {
		effect = {
			every_country = {
				add_ideas = peacetime_training_spirit
			}
		}
	}

	on_weekly = {
		effect = {


			if = {
				limit = {
					has_idea = peacetime_training_spirit
					OR = {
						AND = {
							tag = ITA
							date > 1939.5.31
						}
						AND = {
							OR = {
								tag = JAP
								tag = MAN
								tag = SIA
								original_tag = SPR
							}
							date > 1941.4.1
						}
					}
				}
				remove_ideas = peacetime_training_spirit
			}


			if = {
				limit = {
					has_war = no
					NOT = { has_idea = peacetime_training_spirit }
					OR = {
						NOT = {
							OR = {
								tag = ITA
								tag = JAP
								tag = MAN
								tag = SIA
								original_tag = SPR
							}
						}
						AND = {
							tag = ITA
							date < 1939.5.31
						}
						AND = {
							OR = {
								tag = JAP
								tag = MAN
								tag = SIA
								original_tag = SPR
							}
							date < 1941.4.1
						}
					}
				}
				add_ideas = peacetime_training_spirit
			}
		}
	}
}
