# Manchukuo historical event scheduling + banditry tick.
on_actions = {

	# Monthly construction-MIO funding, scaled by Manchukuo's total civilian factories.
	on_monthly = {
		effect = {
			if = {
				limit = { country_exists = MAN }
				MAN = { MAN_apply_mio_civ_funds = yes }
			}
		}
	}

	# Ruan Zhenduo (Co-Founder of the Concordia Association). While hired, +0.5%/week
	# Factory Output to the Dreams spirit, tracked in MAN_ruan_factory_bonus and capped
	# at +20% (so his share of the shared Dream variable never exceeds 0.20).
	on_weekly = {
		effect = {
			if = {
				limit = {
					tag = MAN
					has_idea_with_trait = MAN_cofounder_of_concordia
					check_variable = { MAN_ruan_factory_bonus < 0.20 }
				}
				add_to_variable = { MAN_ruan_factory_bonus = 0.005 }
				add_to_variable = { MAN_dream_industrial_capacity_factory = 0.005 }
				force_update_dynamic_modifier = yes
			}
		}
	}

	on_startup = {
		effect = {
			# on_startup fires in global scope, so scope explicitly into each country.
			if = {
				limit = { country_exists = MAN }
				MAN = {
					if = {
						limit = { NOT = { has_country_flag = MAN_sys_init } }
						MAN_systems_init = yes
					}
					if = {
						limit = { has_start_date < 1936.1.2 }
						# Lore series
						# 1936
						# 1937
						# 1938+
					}
				}
			}
			if = {
				limit = { country_exists = JAP }
				JAP = {
					if = {
						limit = { has_start_date < 1936.1.2 }
					}
				}
			}
		}
	}


	# Manchukuo systems: Plan, Bureaucracy, Public Security
	on_weekly_MAN = {
		effect = {
			if = {
				limit = { has_country_flag = MAN_sys_init }
				MAN_sec_weekly = yes
				MAN_pol_weekly = yes
				MAN_plan_weekly = yes
				MAN_ui_refresh = yes
			}
		}
	}
	on_monthly_MAN = {
		effect = {
			if = {
				limit = { has_country_flag = MAN_sys_init }
				MAN_pol_monthly = yes
				MAN_sec_monthly = yes
			}
		}
	}
}
