# Precomputes the War in the Pacific UI region-control flags weekly instead of
# evaluating 140 state-controller checks per frame in the scripted GUI triggers.
# The GUI (WAR_IN_PACIFIC_Scripted_Gui.txt) now just reads global.pui_* variables.
on_actions = {
	on_weekly_JAP = {
		effect = {
			pacific_ui_update_region_flags = yes
		}
	}
	on_startup = {
		effect = {
			# on_startup has no country scope; run the update in JAP scope explicitly
			JAP = {
				pacific_ui_update_region_flags = yes
			}
		}
	}
}
