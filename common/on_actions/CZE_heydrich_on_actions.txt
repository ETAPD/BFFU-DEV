on_actions = {
	on_operation_completed = {
		effect = {
			every_operative = {
				limit = { has_trait = CZE_czech_patriot }
				kill_operative = { killed_by = FROM }
			}
		}
	}
	on_game_start = {
		effect = {
			if = {
				limit = { original_tag = CZE }
				set_global_flag = neutralized_focus_CZE_coalition_building
				clr_country_flag = CZE_show_a_divided_nation_category
			}
			if = {
				limit = { original_tag = ENG }
				set_country_flag = ENG_heydrich_costs_v2_initialized
				set_variable = { ENG_heydrich_pp_cost = 50 }
				set_variable = { ENG_heydrich_gun_pp_cost = 25 }
				set_variable = { ENG_heydrich_gun_cost = 2000 }
				set_variable = { ENG_heydrich_gun_threshold = 1999 }
				set_variable = { ENG_heydrich_civ_cost = 1 }
				set_variable = { ENG_heydrich_civ_threshold = 0 }
				set_variable = { ENG_heydrich_countdown = 1826 }
			}
		}
	}
	on_daily = {
		effect = {
			if = {
				limit = { original_tag = ENG }
				if = {
					limit = { NOT = { has_country_flag = ENG_heydrich_costs_v2_initialized } }
					set_country_flag = ENG_heydrich_costs_v2_initialized
					set_variable = { ENG_heydrich_pp_cost = 50 }
					set_variable = { ENG_heydrich_gun_pp_cost = 25 }
					set_variable = { ENG_heydrich_gun_cost = 2000 }
					set_variable = { ENG_heydrich_gun_threshold = 1999 }
					set_variable = { ENG_heydrich_civ_cost = 1 }
					set_variable = { ENG_heydrich_civ_threshold = 0 }
					if = {
						limit = { NOT = { has_variable = ENG_heydrich_countdown } }
						set_variable = { ENG_heydrich_countdown = 1826 }
					}
				}
				if = {
					limit = {
						NOT = { has_country_flag = ENG_heydrich_tickdown_started }
						has_war_with = GER
					}
					set_country_flag = ENG_heydrich_tickdown_started
					set_variable = { ENG_heydrich_countdown = 1826 }
				}
				else_if = {
					limit = {
						has_country_flag = ENG_heydrich_tickdown_started
						NOT = { has_country_flag = ENG_heydrich_operation_unlocked }
					}
					add_to_variable = { ENG_heydrich_countdown = -1 }
					if = {
						limit = { check_variable = { var = ENG_heydrich_countdown value = 0 compare = less_than_or_equals } }
						set_variable = { ENG_heydrich_countdown = 0 }
						set_country_flag = ENG_heydrich_operation_unlocked
					}
				}
			}
			else_if = {
				limit = { original_tag = CZE }
				if = {
					limit = {
						has_completed_focus = CZE_B6_05_HUB_RIGHT
						NOT = { has_completed_focus = CZE_B6R_01_B }
						check_variable = {
							var = CZE_heydrich_delay_bonus
							value = 0.10
							compare = less_than
						}
					}
					add_to_variable = { CZE_heydrich_delay_days = 1 }
					if = {
						limit = {
							check_variable = {
								var = CZE_heydrich_delay_days
								value = 30
								compare = greater_than_or_equals
							}
						}
						set_variable = { CZE_heydrich_delay_days = 0 }
						add_to_variable = { CZE_heydrich_delay_bonus = 0.005 }
						add_to_variable = { CZE_heydrich_delay_stat = 0.5 }
						if = {
							limit = {
								check_variable = {
									var = CZE_heydrich_delay_bonus
									value = 0.10
									compare = greater_than
								}
							}
							set_variable = { CZE_heydrich_delay_bonus = 0.10 }
							set_variable = { CZE_heydrich_delay_stat = 10 }
						}
					}
				}
				if = {
					limit = {
						date > 1936.3.31
						NOT = { has_country_flag = CZE_hossbach_event_applied }
						NOT = { has_completed_focus = CZE_B6_ROOT }
					}
					set_country_flag = CZE_hossbach_event_applied
					country_event = { id = cze_crisis.0 }
				}
				if = {
					limit = { has_country_flag = CZE_show_a_divided_nation_category }
					clr_country_flag = CZE_show_a_divided_nation_category
				}
				if = {
					limit = {
						NOT = { has_country_flag = CZE_anschluss_event_applied }
						GER = { has_country_flag = AUS_annexed }
					}
					set_country_flag = CZE_anschluss_event_applied
					country_event = { id = cze_crisis.1 }
				}
				if = {
					limit = {
						date > 1937.4.30
						NOT = { has_country_flag = CZE_sudeten_demand_event_applied }
						NOT = { has_completed_focus = CZE_B6_01_C }
					}
					set_country_flag = CZE_sudeten_demand_event_applied
					country_event = { id = cze_crisis.2 }
				}
				if = {
					limit = {
						date > 1937.11.30
						NOT = { has_country_flag = CZE_runciman_event_applied }
						NOT = { has_completed_focus = CZE_B6_02_MERGE }
					}
					set_country_flag = CZE_runciman_event_applied
					country_event = { id = cze_crisis.4 }
				}
				if = {
					limit = {
						date > 1938.2.28
						NOT = { has_country_flag = CZE_general_mobilisation_event_applied }
						NOT = { has_completed_focus = CZE_B6_03_A }
					}
					set_country_flag = CZE_general_mobilisation_event_applied
					country_event = { id = cze_crisis.6 }
				}
				if = {
					limit = {
						NOT = { has_country_flag = CZE_sudeten_transfer_event_applied }
						NOT = { has_completed_focus = CZE_B6_03_B }
						GER = { has_country_flag = sudetenland_acquired }
					}
					set_country_flag = CZE_sudeten_transfer_event_applied
					country_event = { id = cze_crisis.5 }
				}
				if = {
					limit = {
						NOT = { has_country_flag = CZE_fate_event_applied }
						GER = { has_completed_focus = GER_fate_of_czechoslovakia }
					}
					set_country_flag = CZE_fate_event_applied
					country_event = { id = cze_crisis.7 }
				}
				if = {
					limit = {
						NOT = { has_country_flag = CZE_protectorate_outcome_applied }
						owns_state = 9
						owns_state = 75
						NOT = { owns_state = 70 }
						NOT = { owns_state = 71 }
					}
					country_event = { id = cze_hist.9 }
				}
				if = {
					limit = {
						NOT = { has_country_flag = CZE_slovak_outcome_applied }
						owns_state = 70
						owns_state = 71
						NOT = { owns_state = 9 }
						NOT = { owns_state = 75 }
					}
					country_event = { id = cze_hist.10 }
				}
			}
		}
	}
}
