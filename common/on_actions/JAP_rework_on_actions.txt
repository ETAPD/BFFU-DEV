on_actions = {

	# Schedule the historical cabinet-transition events at game start.
	on_startup = {
		effect = {
			if = {
				limit = { country_exists = JAP }
				JAP = {
					country_event = { id = JAP_rework.1 days = 56 }   # 26 Feb 1936 - February 26 Incident
					country_event = { id = JAP_rework.2 days = 398 }  # 2 Feb 1937 - Hirota's cabinet falls
					country_event = { id = JAP_rework.3 days = 521 }  # 5 Jun 1937 - Konoe forms his first cabinet (Hayashi out after ~123 days)
					country_event = { id = JAP_rework.4 days = 1100 } # 5 Jan 1939 - Hiranuma cabinet forms
					country_event = { id = JAP_rework.5 days = 1337 } # 30 Aug 1939 - Konoe returns for his second cabinet
					country_event = { id = JAP_rework.6 days = 1827 } # 1 Jan 1941 - Tojo outmaneuvers Konoe
					country_event = { id = JAP_rework.24 days = 1664 } # 22 Jul 1940 - gate The Melancholy Prince to its final stage
					country_event = { id = JAP_rework.30 days = 553 }  # 7 Jul 1937 - Japan opens the war with China (Marco Polo Bridge Incident)
				}
			}
		}
	}


	# Auto-complete Abandon Ties to the West once the Allies (UK/USA) embargo Japan.
	# War tree: Japan capturing a state may complete an automatic focus (Nanjing falls, etc).
	# ROOT is the new controller.
	on_state_control_changed = {
		effect = {
			if = {
				limit = { tag = JAP }
				JAP_war_autocomplete_check = yes
			}
		}
	}

	# RCC advisor portfolios: while seated, each grows one Collaborationist Regime value up to a cap (the gain stays).
	on_weekly_RCC = {
		effect = {
			if = {
				limit = {
					has_idea_with_trait = RCC_master_of_the_purse
					check_variable = { RCC_wk_zhou < 0.1 }
				}
				add_to_variable = { RCC_wk_zhou = 0.0025 }
				add_to_variable = { RCC_pol_ic = 0.0025 }
				force_update_dynamic_modifier = yes
			}
			if = {
				limit = {
					has_idea_with_trait = RCC_southern_literatus
					check_variable = { RCC_wk_liang > -0.08 }
				}
				add_to_variable = { RCC_wk_liang = -0.0025 }
				add_to_variable = { RCC_pol_res = -0.0025 }
				force_update_dynamic_modifier = yes
			}
			if = {
				limit = {
					has_idea_with_trait = RCC_legislative_president
					check_variable = { RCC_wk_chen < 0.08 }
				}
				add_to_variable = { RCC_wk_chen = 0.0025 }
				add_to_variable = { RCC_pol_stab = 0.0025 }
				force_update_dynamic_modifier = yes
			}
			if = {
				limit = {
					has_idea_with_trait = RCC_lord_of_the_north
					check_variable = { RCC_wk_wangyt < 0.1 }
				}
				add_to_variable = { RCC_wk_wangyt = 0.0025 }
				add_to_variable = { RCC_pol_lrf = 0.0025 }
				force_update_dynamic_modifier = yes
			}
			if = {
				limit = {
					has_idea_with_trait = RCC_master_of_no76
					check_variable = { RCC_wk_li > -0.1 }
				}
				add_to_variable = { RCC_wk_li = -0.005 }
				add_to_variable = { RCC_pol_res = -0.005 }
				force_update_dynamic_modifier = yes
			}
			if = {
				limit = {
					has_idea_with_trait = RCC_double_dealer
					check_variable = { RCC_wk_ding < 0.08 }
				}
				add_to_variable = { RCC_wk_ding = 0.0025 }
				add_to_variable = { RCC_pol_rec = 0.0025 }
				force_update_dynamic_modifier = yes
			}
		}
	}

	on_monthly_RCC = {
		effect = {
			# Wang Jingwei dies in November 1944 and Chen Gongbo succeeds him.
			if = {
				limit = {
					date > 1944.11.9
					has_country_flag = RCC_reorganized
					NOT = { has_country_flag = RCC_wang_died }
				}
				country_event = { id = rcc.2 }
			}
			# The bullet lodged in Wang Jingwei's spine since 1935 worsens in stages.
			if = {
				limit = {
					date > 1943.6.1
					has_country_leader = { character = RCC_wang_jingwei ruling_only = yes }
					has_country_leader_with_trait = RCC_bullet_of_1935_1
				}
				remove_country_leader_trait = RCC_bullet_of_1935_1
				add_country_leader_trait = RCC_bullet_of_1935_2
			}
			if = {
				limit = {
					date > 1944.3.1
					has_country_leader = { character = RCC_wang_jingwei ruling_only = yes }
					has_country_leader_with_trait = RCC_bullet_of_1935_2
				}
				remove_country_leader_trait = RCC_bullet_of_1935_2
				add_country_leader_trait = RCC_bullet_of_1935_3
			}
			# September 1943: Li Shiqun is poisoned by the Kempeitai.
			if = {
				limit = {
					date > 1943.9.6
					has_country_flag = RCC_reorganized
					NOT = { has_country_flag = RCC_li_shiqun_poisoned }
				}
				country_event = { id = rcc.3 }
			}
		}
	}

	on_weekly_JAP = {
		effect = {
			if = { limit = { has_country_flag = JAP_cb_active } JAP_cb_weekly = yes }
			JAP_war_autocomplete_check = yes
			if = {
				limit = {
					has_completed_focus = JAP_hiranuma_cabinet_forms
					NOT = { has_completed_focus = JAP_abandon_ties_to_the_west }
					OR = { is_embargoed_by = ENG is_embargoed_by = USA }
				}
				complete_national_focus = JAP_abandon_ties_to_the_west
			}
		}
	}

	# Keep the Soviet non-aggression truce alive while the pact stands. Renewed monthly at 45 days
	# (always longer than the month between checks), so it never lapses until the break decision
	# clears the flag - then it expires on its own within ~6 weeks.
	on_monthly_JAP = {
		effect = {
			if = {
				limit = {
					has_country_flag = JAP_nap_with_SOV
					country_exists = SOV
				}
				set_truce = { target = SOV days = 45 }
			}
			# Recompute Pacific objectives (Tora trim, Festung Europa, Allied diversion).
			if = {
				limit = { has_country_flag = JAP_pacific_war_active }
				JAP_pacific_recompute = yes
			}
			# Everything for the War Effort (Tojo Outmaneuvers Konoe): the home
			# islands' civilian economy keeps bleeding into military production.
			# Converts 2 civilian factories a month, one at a time so a state that
			# runs dry doesn't block the other conversion.
			if = {
				limit = { has_country_flag = JAP_everything_for_war_effort }
				random_owned_state = {
					limit = {
						JAP_is_home_islands_state = yes
						industrial_complex > 0
					}
					add_building_construction = { type = industrial_complex level = -1 instant_build = yes }
					add_building_construction = { type = arms_factory level = 1 instant_build = yes }
				}
				random_owned_state = {
					limit = {
						JAP_is_home_islands_state = yes
						industrial_complex > 0
					}
					add_building_construction = { type = industrial_complex level = -1 instant_build = yes }
					add_building_construction = { type = arms_factory level = 1 instant_build = yes }
				}
			}
			# Matsumoto's flat monthly market injection (imported from the Benjumea mechanic).
			if = {
				limit = { has_idea_with_trait = JAP_yasukawa_zaibatsu_president }
				add_cic = 1500
			}
			# Matsumoto's 5% monthly chance of a speculative loss.
			if = {
				limit = { has_idea_with_trait = JAP_yasukawa_zaibatsu_president }
				random_list = {
					1 = { country_event = { id = JAP_rework.41 hours = 6 } }
					19 = {}
				}
			}
			# Kuhara's 5% monthly chance of a banking windfall.
			if = {
				limit = { has_idea_with_trait = JAP_financial_ally }
				random_list = {
					1 = { country_event = { id = JAP_rework.42 hours = 6 } }
					19 = {}
				}
			}
		}
	}

	# When China capitulates, lift Imperial Directive 501 (the escalation spirit).
	on_capitulation = {
		effect = {
			if = {
				limit = { tag = CHI }
				JAP = {
					if = {
						limit = { has_idea = JAP_imperial_directive_501 }
						remove_ideas = JAP_imperial_directive_501
					}
				}
			}
		}
	}
}
