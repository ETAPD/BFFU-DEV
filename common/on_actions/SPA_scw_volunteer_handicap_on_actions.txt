on_actions = {
	on_send_volunteers = {
		effect = {
			if = {
				limit = {
					has_global_flag = spanish_civil_war
					NOT = { has_global_flag = scw_over }
					NOT = { original_tag = SPR }
					NOT = { has_idea = SPA_scw_nationalist_handicap }
					FROM = {
						has_country_flag = SPA_scw_nationalist_side
					}
				}
				add_ideas = SPA_scw_nationalist_handicap
			}
		}
	}

	# Safety net: if the Nationalists ever end up flagged but without the spirit
	# (focus/event swapped their spirits, or the war began before this system
	# existed
	on_weekly = {
		effect = {
			if = {
				limit = {
					has_country_flag = SPA_scw_nationalist_side
					has_global_flag = spanish_civil_war
					NOT = { has_global_flag = scw_over }
					NOT = { has_idea = SPA_scw_nationalist_handicap }
				}
				add_ideas = SPA_scw_nationalist_handicap
			}
		}
	}
}
