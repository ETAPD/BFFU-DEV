on_actions = {
	on_startup = {
		effect = {
			every_country = {
				if = {
					limit = {
						is_ai = no
						NOT = { has_country_flag = bffu_se_init }
					}
					set_variable = { bffu_superevents_enabled = 2 }
					set_country_flag = bffu_se_init
				}
			}
		}
	}

	on_declare_war = {
		effect = {
			if = {
				limit = {
					ROOT = { tag = SPR }
					FROM = { tag = D01 }
				}
				every_country = {
					limit = { is_ai = no }
					country_event = { id = bffu_super.1 }
				}
			}
			if = {
				limit = {
					ROOT = { tag = JAP }
					FROM = { tag = CHI }
				}
				every_country = {
					limit = { is_ai = no }
					country_event = { id = bffu_super.2 }
				}
			}
			if = {
				limit = {
					ROOT = { tag = GER }
					FROM = { tag = POL }
				}
				every_country = {
					limit = { is_ai = no }
					country_event = { id = bffu_super.3 }
				}
			}
			if = {
				limit = {
					ROOT = { tag = JAP }
					FROM = { tag = USA }
				}
				every_country = {
					limit = { is_ai = no }
					country_event = { id = bffu_super.4 }
				}
			}
			if = {
				limit = {
					ROOT = { tag = GER }
					FROM = { tag = SOV }
				}
				every_country = {
					limit = { is_ai = no }
					country_event = { id = bffu_super.5 }
				}
			}
		}
	}

	on_capitulation = {
		effect = {
			if = {
				limit = {
					ROOT = { tag = ENG }
				}
				every_country = {
					limit = { is_ai = no }
					country_event = { id = bffu_super.8 }
				}
			}
		}
	}
}
