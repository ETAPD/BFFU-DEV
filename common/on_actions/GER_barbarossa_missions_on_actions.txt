# Daily-checked replacement 

on_actions = {
	on_daily = {
		effect = {
			if = {
				limit = {
					tag = GER
					has_country_flag = GER_barbarossa_missions_active_flag
				}
				add_to_variable = { GER_barbarossa_mission_days_elapsed = 1 }
				if = {
					limit = {
						check_variable = { GER_barbarossa_mission_days_elapsed > 119 }
					}
					clr_country_flag = GER_barbarossa_missions_active_flag
					country_event = { id = germany.300 days = 1 }
					country_event = { id = germany.301 days = 1 }
					country_event = { id = germany.302 days = 1 }
				}
			}
		}
	}
}
