on_actions = {
	on_government_exiled = {
		effect = {
			#transfer INS to AST
			if = {
				limit = {
					original_tag = HOL
					has_country_flag = HOL_australia_can_annex_indonesia
					NOT = { has_country_flag = HOL_already_annexed_dutch_colonies }
				}
				HOL = { set_country_flag = HOL_already_annexed_dutch_colonies }
				INS = {
					every_unit_leader = {
						set_nationality = HOL
					}
				}
				HOL = { set_capital = { state = 309 } }
				HOL = {
					set_country_flag = HOL_exile_transfer_to_ast
					country_event = { id = mtg_netherlands.100 hours = 1 }
				}
			}
			#transfer INS to HOL
			if = {
				limit = {
					original_tag = HOL
					has_country_flag = HOL_netherlands_can_annex_indonesia
					NOT = { has_country_flag = HOL_already_annexed_dutch_colonies }
				}
				HOL = { set_country_flag = HOL_already_annexed_dutch_colonies }
				INS = {
					every_unit_leader = {
						set_nationality = HOL
					}
				}
				INS = {
					hidden_effect = {
						country_event = { id = mtg_netherlands.22 }
					}
				}
				hidden_effect = { news_event = { id = mtg_news.46 days = 1 } }
				HOL = { set_capital = { state = 335 } }
				HOL = {
					country_event = { id = mtg_netherlands.100 hours = 1 }
				}
			}
			if = {
				limit = {
					original_tag = HOL
					NOT = { has_country_flag = HOL_netherlands_can_annex_indonesia }
					NOT = { has_country_flag = HOL_australia_can_annex_indonesia }
					NOT = { has_country_flag = HOL_already_annexed_dutch_colonies }
				}
				HOL = { set_country_flag = HOL_already_annexed_dutch_colonies }
				INS = {
					every_unit_leader = {
						set_nationality = HOL
					}
				}
				HOL = { set_capital = { state = 309 } }
				HOL = {
					set_country_flag = HOL_exile_transfer_to_ast
					country_event = { id = mtg_netherlands.100 hours = 1 }
				}
			}
		}
	}
}
