on_actions = {
	on_monthly_CZE = {
		effect = {
			if = {
				limit = {
					has_completed_focus = CZE_protectorate_proclaimed
					NOT = { has_completed_focus = CZE_appoint_reinhard }
					check_variable = { CZE_heydrich_delay_bonus < 0.10 }
				}
				add_to_variable = { CZE_heydrich_delay_bonus = 0.005 }
				clamp_variable = { var = CZE_heydrich_delay_bonus min = 0 max = 0.10 }
			}

			# Munich crisis events
			if = {
				limit = {
					date > 1936.3.31
					NOT = { has_country_flag = CZE_hossbach_event_applied }
					NOT = { has_completed_focus = CZE_hossbach_conference }
				}
				set_country_flag = CZE_hossbach_event_applied
				country_event = { id = cze_crisis.0 }
			}
			else_if = {
				limit = {
					NOT = { has_country_flag = CZE_anschluss_event_applied }
					NOT = { has_completed_focus = CZE_anschluss }
					GER = { has_country_flag = AUS_annexed }
				}
				set_country_flag = CZE_anschluss_event_applied
				country_event = { id = cze_crisis.1 }
			}
			else_if = {
				limit = {
					date > 1937.4.30
					NOT = { has_country_flag = CZE_sudeten_demand_event_applied }
					NOT = { has_completed_focus = CZE_may_crisis }
				}
				set_country_flag = CZE_sudeten_demand_event_applied
				country_event = { id = cze_crisis.2 }
			}
			else_if = {
				limit = {
					date > 1937.11.30
					NOT = { has_country_flag = CZE_runciman_event_applied }
					NOT = { has_completed_focus = CZE_runciman_report }
				}
				set_country_flag = CZE_runciman_event_applied
				country_event = { id = cze_crisis.4 }
			}
			else_if = {
				limit = {
					date > 1938.2.28
					NOT = { has_country_flag = CZE_general_mobilisation_event_applied }
					NOT = { has_completed_focus = CZE_general_mobilisation }
				}
				set_country_flag = CZE_general_mobilisation_event_applied
				country_event = { id = cze_crisis.6 }
			}
			else_if = {
				limit = {
					NOT = { has_country_flag = CZE_sudeten_transfer_event_applied }
					NOT = { has_completed_focus = CZE_munich_agreement }
					GER = { has_country_flag = sudetenland_acquired }
				}
				set_country_flag = CZE_sudeten_transfer_event_applied
				country_event = { id = cze_crisis.5 }
			}
			else_if = {
				limit = {
					NOT = { has_country_flag = CZE_fate_event_applied }
					NOT = { has_completed_focus = CZE_wehrmacht_march }
					GER = { has_completed_focus = GER_fate_of_czechoslovakia }
				}
				set_country_flag = CZE_fate_event_applied
				country_event = { id = cze_crisis.7 }
			}

			# Fallback
			if = {
				limit = {
					NOT = { has_country_flag = CZE_protectorate_outcome_applied }
					NOT = { has_country_flag = CZE_slovak_outcome_applied }
					owns_state = 9
					owns_state = 75
					NOT = { owns_state = 70 }
					NOT = { owns_state = 71 }
				}
				country_event = { id = cze_hist.9 }
			}
			else_if = {
				limit = {
					NOT = { has_country_flag = CZE_protectorate_outcome_applied }
					NOT = { has_country_flag = CZE_slovak_outcome_applied }
					owns_state = 70
					owns_state = 71
					NOT = { owns_state = 9 }
					NOT = { owns_state = 75 }
				}
				country_event = { id = cze_hist.10 }
			}
		}
	}
}
