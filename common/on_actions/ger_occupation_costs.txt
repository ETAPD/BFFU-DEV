    on_weekly = {
        effect = {
            if = {
                limit = { tag = GER } # Make sure you are testing as Germany!
                
                # 1. Reset the tracker variable on the country
                set_variable = { global_occupied_civs = 0 }

                # 2. Iterate through every state you control
                every_controlled_state = {
                    # Filter: Only pull states that are NOT your native cores
                    limit = {
                        NOT = { is_core_of = ROOT }
                    }

                    # 3. Read the state's civilian factories and add them directly to ROOT
                    if = {
                        limit = { 
                            has_building = industrial_complex 
                        }
                        # Explictly targets the root country tag to stack the values safely
                        ROOT = {
                            add_to_variable = { 
                                global_occupied_civs = PREV.num_of_civilian_factories 
                            }
                        }
                    }
                }
            }
        }
    }
