on_actions = {
	on_startup = {
		effect = {
			OBG = { fuel_startup_effect = yes }
		}
	}

	on_daily_OBG = {
		effect = {
			if = {
				limit = { NOT = { has_global_flag = fuel_manager_reserve_fixed } }
				if = { limit = { country_exists = GER } GER = { set_variable = { var = min_fuel value = 0.05 } } }
				if = { limit = { country_exists = ENG } ENG = { set_variable = { var = min_fuel value = 0.05 } } }
				if = { limit = { country_exists = JAP } JAP = { set_variable = { var = min_fuel value = 0.05 } } }
				set_global_flag = fuel_manager_reserve_fixed
			}
			on_daily_fuel = yes
		}
	}

	on_monthly_OBG = {
		effect = {
			# Keep every manager's member list up to date (AI and players alike). The refresh only
			# rebuilds when the membership changed, so a player's priority order survives otherwise.
			for_each_scope_loop = {
				array = GLOBAL.fuel_system_manager    # GER, ENG, JAP - THIS = that manager
				fuel_system_refresh_list = yes
			}
		}
	}
}
