on_actions = {

	# Build the fuel-manager list (GER/JAP/ENG), each manager's faction_members_array
	# and the GLOBAL.fuel_countries array, and apply the miss-nothing min_fuel default.
	on_startup = {
		effect = {
			SWE = { fuel_startup_effect = yes }
		}
	}

	# Run the fuel redistribution once per day. on_daily_SOV fires only in SOV scope
	# (a permanent major), so on_daily_fuel runs exactly once per day rather than once
	# per country. The effect operates on the GLOBAL fuel arrays, so the scope only
	# needs to be a country that exists. (Mirrors Horst's on_daily_SOV wiring.)
	on_daily_SOV = {
		effect = {
			if = {
				limit = { NOT = { has_global_flag = fuel_manager_reserve_fixed } }
				if = { limit = { country_exists = GER } GER = { set_variable = { var = min_fuel value = 0.05 } } }
				if = { limit = { country_exists = ENG } ENG = { set_variable = { var = min_fuel value = 0.05 } } }
				if = { limit = { country_exists = JAP } JAP = { set_variable = { var = min_fuel value = 0.05 } } }
				set_global_flag = fuel_manager_reserve_fixed
			}
			on_daily_fuel = yes
		}
	}
}
