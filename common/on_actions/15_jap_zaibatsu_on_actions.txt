on_actions = {
	on_weekly_JAP = {
		effect = {
			if = {
				limit = { has_country_flag = JAP_zaibatsu_economy_active }

			# --- Big Four path: Nissan and Nakajima are permanently obstructive ---
			if = {
				limit = { has_country_flag = JAP_big_four_path }
				set_variable = { JAP_zaibatsu_nissan_favor = 0 }
				set_variable = { JAP_zaibatsu_nakajima_favor = 0 }
			}

			# --- obstruction when favor collapses to 0 ---
			if = {
				limit = {
					check_variable = { JAP_zaibatsu_mitsui_favor < 1 }
					NOT = { has_country_flag = JAP_zaibatsu_mitsui_obstruction_flag }
					NOT = { has_country_flag = JAP_zaibatsu_cartel_flag }
				}
				country_event = { id = jap_zaibatsu.21 hours = 6 }
			}
			if = {
				limit = {
					check_variable = { JAP_zaibatsu_mitsubishi_favor < 1 }
					NOT = { has_country_flag = JAP_zaibatsu_mitsubishi_obstruction_flag }
					NOT = { has_country_flag = JAP_zaibatsu_cartel_flag }
				}
				country_event = { id = jap_zaibatsu.22 hours = 6 }
			}
			if = {
				limit = {
					check_variable = { JAP_zaibatsu_sumitomo_favor < 1 }
					NOT = { has_country_flag = JAP_zaibatsu_sumitomo_obstruction_flag }
					NOT = { has_country_flag = JAP_zaibatsu_cartel_flag }
				}
				country_event = { id = jap_zaibatsu.23 hours = 6 }
			}

				limit = {
					check_variable = { JAP_zaibatsu_yasuda_favor < 1 }
					NOT = { has_country_flag = JAP_zaibatsu_yasuda_obstruction_flag }
					NOT = { has_country_flag = JAP_zaibatsu_cartel_flag }
				}
				country_event = { id = jap_zaibatsu.24 hours = 6 }

				limit = {
					check_variable = { JAP_zaibatsu_nissan_favor < 1 }
					NOT = { has_country_flag = JAP_zaibatsu_nissan_obstruction_flag }
					NOT = { has_country_flag = JAP_zaibatsu_cartel_flag }
				}
				country_event = { id = jap_zaibatsu.25 hours = 6 }
			}
			if = {
				limit = {
					check_variable = { JAP_zaibatsu_nakajima_favor < 1 }
					NOT = { has_country_flag = JAP_zaibatsu_nakajima_obstruction_flag }
					NOT = { has_country_flag = JAP_zaibatsu_cartel_flag }
				}
				country_event = { id = jap_zaibatsu.26 hours = 6 }
			}

			# --- favor 50 rewards ---
			if = {
				limit = {
					check_variable = { JAP_zaibatsu_mitsui_favor > 49 }
					NOT = { has_country_flag = JAP_zaibatsu_mitsui_reward50_flag }
				}
				country_event = { id = jap_zaibatsu.31 hours = 6 }
			}
			if = {
				limit = {
					check_variable = { JAP_zaibatsu_mitsubishi_favor > 49 }
					NOT = { has_country_flag = JAP_zaibatsu_mitsubishi_reward50_flag }
				}
				country_event = { id = jap_zaibatsu.32 hours = 6 }
			}
			if = {
				limit = {
					check_variable = { JAP_zaibatsu_sumitomo_favor > 49 }

				}

			}
			if = {
				limit = {
					check_variable = { JAP_zaibatsu_nissan_favor > 49 }

					NOT = { has_country_flag = JAP_zaibatsu_nakajima_reward50_flag }
				}
				country_event = { id = jap_zaibatsu.36 hours = 6 }
			}

			if = {
				limit = {
					check_variable = { JAP_zaibatsu_mitsui_favor > 99 }
					NOT = { has_country_flag = JAP_zaibatsu_mitsui_reward100_flag }
				}

				limit = {
					check_variable = { JAP_zaibatsu_yasuda_favor > 99 }
					NOT = { has_country_flag = JAP_zaibatsu_yasuda_reward100_flag }
				}

					NOT = { has_country_flag = JAP_zaibatsu_nissan_reward100_flag }
				}

			}
			if = {
				limit = {
					check_variable = { JAP_zaibatsu_nakajima_favor > 99 }
					NOT = { has_country_flag = JAP_zaibatsu_nakajima_reward100_flag }
				}
				country_event = { id = jap_zaibatsu.46 hours = 6 }
			}

			if = {

					NOT = { has_country_flag = JAP_zaibatsu_demand_c1_resolved }

			}

				limit = {
					date > 1940.6.1
					NOT = { has_country_flag = JAP_zaibatsu_demand_c3_resolved }

				set_country_flag = JAP_zaibatsu_demand_c3_resolved

					limit = { has_idea = JAP_new_economic_structure }
					country_event = { id = jap_zaibatsu.54 hours = 6 }
				}
				else = {
					country_event = { id = jap_zaibatsu.64 hours = 6 }
				}
			}
			if = {
				limit = {
					date > 1939.1.1
					NOT = { has_country_flag = JAP_zaibatsu_demand_c5_resolved }
				}
				set_country_flag = JAP_zaibatsu_demand_c5_resolved
				if = {
					limit = {
						OR = {
							has_idea = JAP_yen_block
							has_idea = JAP_yen_block_tripartite
						}
					}
					country_event = { id = jap_zaibatsu.55 hours = 6 }
				}
				else = {
					country_event = { id = jap_zaibatsu.65 hours = 6 }
				}
			}
			if = {
				limit = {
					date > 1940.8.1
					NOT = { has_country_flag = JAP_zaibatsu_demand_c6_resolved }
				}
				set_country_flag = JAP_zaibatsu_demand_c6_resolved
				if = {
					limit = { num_of_military_factories > 299 }
					country_event = { id = jap_zaibatsu.56 hours = 6 }
				}
				else = {
					country_event = { id = jap_zaibatsu.66 hours = 6 }
				}
			}
			if = {
				limit = {
					date > 1940.1.1
					NOT = { has_country_flag = JAP_zaibatsu_demand_c14_resolved }
				}
				set_country_flag = JAP_zaibatsu_demand_c14_resolved

					limit = { MAN = { num_of_factories > 199 } }
					country_event = { id = jap_zaibatsu.57 hours = 6 }
				}
				else = {
					country_event = { id = jap_zaibatsu.67 hours = 6 }
				}
			}
			if = {
				limit = {
					date > 1939.6.1
					NOT = { has_country_flag = JAP_zaibatsu_demand_c15_resolved }
				}
				set_country_flag = JAP_zaibatsu_demand_c15_resolved
				if = {
					limit = {
						OR = {
							mio:JAP_nakajima_organization = { has_mio_size > 2 }
							has_tech = excavation3
						}
					}
					country_event = { id = jap_zaibatsu.58 hours = 6 }
				}
				else = {
					country_event = { id = jap_zaibatsu.68 hours = 6 }
				}
			}
			}

			# --- Everything for the War Effort: total-war footing spreads across
			# whatever land is currently conquered (Tojo Outmaneuvers Konoe) ---
			if = {
				limit = { has_country_flag = JAP_everything_for_war_effort }

					limit = { NOT = { is_core_of = JAP } }
					add_dynamic_modifier = { modifier = JAP_everything_for_war_effort_dm }
