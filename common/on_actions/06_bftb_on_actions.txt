	on_civil_war_end = {
		effect = {

				}
				set_country_flag = achievement_greek_civility

			if = {
				limit = {
					any_country = {

							has_country_flag = TUR_iraqi_government_flag

						}
					}
				}
				TUR = {  
					create_wargoal = {
						type = puppet_wargoal_focus
						target = IRQ
					}
				}
			}
			else_if = {
				limit = {
					any_country = { 
						AND = { 
							has_country_flag = TUR_iraqi_fascists_flag 
							exists = yes
						}
					}
				}
				ENG = { 
					remove_resource_rights = 676
					remove_resource_rights = 291
				}
				TUR = { country_event = { id = bftb_turkey.69 hours = 6 } } 
				GER = { country_event = { id = bftb_turkey.71 hours = 8 } } 
				ITA = { country_event = { id = bftb_turkey.73 hours = 8 } } 
				if = {
					limit = {
						IRQ = { NOT = { has_idea = nationalism } }
					}
					add_ideas = nationalism
				}
			}
		}
		effect = { #SCW
			if = {
				limit = {
					FROM = {
						original_tag = SPR
					}
					NOT = {
						any_country_with_original_tag = {
							original_tag_to_check = SPR
							NOT = { tag = PREV }
							NOT = {
								tag = ROOT
								tag = FROM
							}
							exists = yes
						}
					}
				}
				# ważne
				if = {
					limit = {
						YUG = { has_country_flag = YUG_sent_volunteers_to_SPA_flag }
					}
					YUG = {						
						country_event = { id = bftb_bulgaria_scw_volunteers.3 hours = 12 random_hours = 8 }
						clr_country_flag = YUG_sent_volunteers_to_SPA_flag
					}
				}
			}
		}
	}

	# called when a country send volunteers to another
	# ROOT is sender, FROM is receiver
	on_send_volunteers = {
		effect = {
			if = { #Yugoslavia has sent volunteers to Spanish Republicans/Nationalists (LaR)
				limit = {
					tag = YUG
					FROM = { original_tag = SPR }
					has_dlc = "La Resistance"
					SPR_scw_in_progress = yes
				}
				if = {
					limit = {
						FROM = { tag = SPD }
					}
					set_country_flag = YUG_sent_volunteers_to_SPD_flag
				}
				else = {
					set_country_flag = YUG_sent_volunteers_to_SPA_flag
				}
			}
			else_if = {
				limit = {
					tag = YUG
					FROM = { original_tag = SPR }
					NOT = { has_dlc = "La Resistance" }
					FROM = { has_civil_war = yes }
				}
				if = {
					limit = {
						FROM = {
							OR = {
								has_government = democratic
								has_government = communism
							}
						}
					}
					set_country_flag = YUG_sent_volunteers_to_SPD_flag
				}
				else = {
					set_country_flag = YUG_sent_volunteers_to_SPA_flag
				}
			}	
		}
	}
