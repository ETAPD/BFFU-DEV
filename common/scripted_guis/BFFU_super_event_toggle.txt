scripted_gui = {
	bffu_superevents_checkbox_ui = {
		context_type = player_context
		window_name = "bffu_superevents_checkbox_window"
		parent_window_name = "event_header"
		visible = {
			always = yes
		}
		effects = {
			bffu_superevents_checkbox_click = {
				if = {
					limit = {
						ROOT = { NOT = { check_variable = { bffu_superevents_enabled > 0 } } }
					}
					ROOT = { set_variable = { bffu_superevents_enabled = 2 } }
				}
				else = {
					ROOT = { clear_variable = bffu_superevents_enabled }
				}
			}
		}
		triggers = {
			bffu_superevents_checkbox_click_enabled = {
				always = yes
			}
		}
		properties = {
			bffu_superevents_checkbox = {
				frame = bffu_superevents_enabled
			}
		}
	}
}
