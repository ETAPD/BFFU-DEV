# Sphere Ledger texts
defined_text = {
	name = GetJAPSLBalance
	text = {
		trigger = { NOT = { has_country_flag = JAP_sl_active } }
		localization_key = JAP_SL_BAL_NONE
	}
	text = {
		trigger = { check_variable = { JAP_sl_balance > -0.01 } }
		localization_key = JAP_SL_BAL_SURPLUS
	}
	text = {
		localization_key = JAP_SL_BAL_SHORT
	}
}
defined_text = {
	name = GetJAPSLStatus_man
	text = {
		trigger = { NOT = { JAP_sl_active_man = yes } }
		localization_key = JAP_SL_ST_OUT
	}
	text = {
		trigger = { has_country_flag = JAP_sl_granted_man }
		localization_key = JAP_SL_ST_ASSOC
	}
	text = {
		trigger = { check_variable = { JAP_sl_st_man = 1 } }
		localization_key = JAP_SL_ST_REQ
	}
	text = {
		trigger = { check_variable = { JAP_sl_st_man = 3 } }
		localization_key = JAP_SL_ST_PLEDGE_MAN
	}
	text = {
		localization_key = JAP_SL_ST_DEV_MAN
	}
}
