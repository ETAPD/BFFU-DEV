JAP_sl_weekly = {
	JAP_occ_stance_states = yes
	# shipping losses follow the Pacific tide
	if = {
		limit = { has_country_flag = JAP_pacific_war_active }
		if = { limit = { JAP_pacific_tide_winning = yes } set_variable = { JAP_sl_loss = 0.05 } }
		else_if = { limit = { JAP_pacific_tide_losing = yes } set_variable = { JAP_sl_loss = 0.35 } }
		else = { set_variable = { JAP_sl_loss = 0.15 } }
	}
	else = { set_variable = { JAP_sl_loss = 0.02 } }
	if = { limit = { has_country_flag = JAP_sl_escorts } subtract_from_variable = { JAP_sl_loss = 0.10 } }
	clamp_variable = { var = JAP_sl_loss min = 0 max = 1 }
	# imperial demand grows with the wars
	set_variable = { JAP_sl_demand = 15 }
	if = { limit = { has_war_with = CHI } add_to_variable = { JAP_sl_demand = 20 } }
	if = { limit = { has_country_flag = JAP_pacific_war_active } add_to_variable = { JAP_sl_demand = 45 } }
	set_variable = { JAP_sl_delivered = JAP_sl_levy }
	# Manchukuo
	if = {
		limit = { JAP_sl_active_man = yes }
		set_temp_variable = { m = 1 }
		if = { limit = { check_variable = { JAP_sl_st_man = 1 } } set_temp_variable = { m = 1.5 } }
		if = { limit = { check_variable = { JAP_sl_st_man = 2 } } set_temp_variable = { m = 0.8 } }
		if = { limit = { check_variable = { JAP_sl_st_man = 3 } } set_temp_variable = { m = 0.6 } }
		if = { limit = { check_variable = { JAP_sl_st_man = 4 } } set_temp_variable = { m = 0.6 } }
		set_temp_variable = { dv = JAP_sl_dev_man }
		add_to_temp_variable = { dv = 1 }
		set_variable = { JAP_sl_y_man = 10 }
		multiply_variable = { JAP_sl_y_man = m }
		multiply_variable = { JAP_sl_y_man = dv }
		add_to_variable = { JAP_sl_delivered = JAP_sl_y_man }
		if = { limit = { check_variable = { JAP_sl_st_man = 1 } } add_to_variable = { JAP_sl_accord = -0.75 } }
		if = { limit = { check_variable = { JAP_sl_st_man = 2 } } add_to_variable = { JAP_sl_accord = 0.1 } }
		if = { limit = { check_variable = { JAP_sl_st_man = 3 } } add_to_variable = { JAP_sl_accord = 0.5 } }
		if = { limit = { check_variable = { JAP_sl_st_man = 4 } } add_to_variable = { JAP_sl_accord = 0.25 } }
		if = {
			limit = { check_variable = { JAP_sl_st_man = 2 } check_variable = { JAP_sl_dev_man < 0.5 } }
			add_to_variable = { JAP_sl_dev_man = 0.01 }
			if = { limit = { has_country_flag = JAP_sl_ministry } add_to_variable = { JAP_sl_dev_man = 0.005 } }
			clamp_variable = { var = JAP_sl_dev_man min = 0 max = 0.5 }
		}
		if = { limit = { check_variable = { JAP_sl_st_man = 3 } } add_to_variable = { JAP_sl_pl_man = 1 } }
	}
	else = { set_variable = { JAP_sl_y_man = 0 } }
	# North China
	if = {
		limit = { JAP_sl_active_nch = yes }
		set_temp_variable = { m = 1 }
		set_variable = { JAP_sl_rs = 0 }
		set_variable = { JAP_sl_rc = 0 }
		every_state = {
			limit = { JAP_sl_state_in_nch = yes JAP_occ_held = yes }
			add_to_variable = { ROOT.JAP_sl_rs = THIS.resistance }
			add_to_variable = { ROOT.JAP_sl_rc = 1 }
		}
		set_variable = { JAP_sl_res_nch = 0 }
		if = {
			limit = { check_variable = { JAP_sl_rc > 0 } }
			set_variable = { JAP_sl_res_nch = JAP_sl_rs }
			divide_variable = { JAP_sl_res_nch = JAP_sl_rc }
		}
		# resistance eats the yield: -1% per 1.5% average resistance (75% resistance halves it)
		set_temp_variable = { rk = JAP_sl_res_nch }
		divide_temp_variable = { rk = 150 }
		set_temp_variable = { rkeep = 1 }
		subtract_from_temp_variable = { rkeep = rk }
		clamp_temp_variable = { var = rkeep min = 0.2 max = 1 }
		if = { limit = { check_variable = { JAP_sl_st_nch = 1 } } set_temp_variable = { m = 1.5 } }
		if = { limit = { check_variable = { JAP_sl_st_nch = 2 } } set_temp_variable = { m = 0.8 } }
		if = { limit = { check_variable = { JAP_sl_st_nch = 3 } } set_temp_variable = { m = 0.6 } }
		if = { limit = { check_variable = { JAP_sl_st_nch = 4 } } set_temp_variable = { m = 0.6 } }
		set_temp_variable = { dv = JAP_sl_dev_nch }
		add_to_temp_variable = { dv = 1 }
		set_variable = { JAP_sl_y_nch = 9 }
		multiply_variable = { JAP_sl_y_nch = m }
		multiply_variable = { JAP_sl_y_nch = dv }
		multiply_variable = { JAP_sl_y_nch = rkeep }
		add_to_variable = { JAP_sl_delivered = JAP_sl_y_nch }
		if = { limit = { check_variable = { JAP_sl_st_nch = 1 } } add_to_variable = { JAP_sl_accord = -0.75 } }
		if = { limit = { check_variable = { JAP_sl_st_nch = 2 } } add_to_variable = { JAP_sl_accord = 0.1 } }
		if = { limit = { check_variable = { JAP_sl_st_nch = 3 } } add_to_variable = { JAP_sl_accord = 0.5 } }
		if = { limit = { check_variable = { JAP_sl_st_nch = 4 } } add_to_variable = { JAP_sl_accord = 0.25 } }
		if = {
			limit = { check_variable = { JAP_sl_st_nch = 2 } check_variable = { JAP_sl_dev_nch < 0.5 } }
			add_to_variable = { JAP_sl_dev_nch = 0.01 }
			if = { limit = { has_country_flag = JAP_sl_ministry } add_to_variable = { JAP_sl_dev_nch = 0.005 } }
			clamp_variable = { var = JAP_sl_dev_nch min = 0 max = 0.5 }
		}
		if = { limit = { check_variable = { JAP_sl_st_nch = 3 } } add_to_variable = { JAP_sl_pl_nch = 1 } }
	}
	else = { set_variable = { JAP_sl_y_nch = 0 } }
	# Korea and Taiwan
	if = {
		limit = { JAP_sl_active_kor = yes }
		set_temp_variable = { m = 1 }
		if = { limit = { check_variable = { JAP_sl_st_kor = 1 } } set_temp_variable = { m = 1.5 } }
		if = { limit = { check_variable = { JAP_sl_st_kor = 2 } } set_temp_variable = { m = 0.8 } }
		if = { limit = { check_variable = { JAP_sl_st_kor = 3 } } set_temp_variable = { m = 0.6 } }
		if = { limit = { check_variable = { JAP_sl_st_kor = 4 } } set_temp_variable = { m = 0.6 } }

		add_to_temp_variable = { dv = 1 }
		set_variable = { JAP_sl_y_kor = 8 }
		multiply_variable = { JAP_sl_y_kor = m }
		multiply_variable = { JAP_sl_y_kor = dv }
		add_to_variable = { JAP_sl_delivered = JAP_sl_y_kor }
		if = { limit = { check_variable = { JAP_sl_st_kor = 1 } } add_to_variable = { JAP_sl_accord = -0.75 } }
		if = { limit = { check_variable = { JAP_sl_st_kor = 2 } } add_to_variable = { JAP_sl_accord = 0.1 } }
		if = { limit = { check_variable = { JAP_sl_st_kor = 3 } } add_to_variable = { JAP_sl_accord = 0.5 } }
		if = { limit = { check_variable = { JAP_sl_st_kor = 4 } } add_to_variable = { JAP_sl_accord = 0.25 } }
		if = {
			limit = { check_variable = { JAP_sl_st_kor = 2 } check_variable = { JAP_sl_dev_kor < 0.5 } }
			add_to_variable = { JAP_sl_dev_kor = 0.01 }
			if = { limit = { has_country_flag = JAP_sl_ministry } add_to_variable = { JAP_sl_dev_kor = 0.005 } }
			clamp_variable = { var = JAP_sl_dev_kor min = 0 max = 0.5 }
		}
		if = { limit = { check_variable = { JAP_sl_st_kor = 3 } } add_to_variable = { JAP_sl_pl_kor = 1 } }
	}
	else = { set_variable = { JAP_sl_y_kor = 0 } }
	# Indochina
	if = {
		limit = { JAP_sl_active_inc = yes }
		set_temp_variable = { m = 1 }
		set_variable = { JAP_sl_rs = 0 }
		set_variable = { JAP_sl_rc = 0 }
		every_state = {
			limit = { JAP_sl_state_in_inc = yes JAP_occ_held = yes }
			add_to_variable = { ROOT.JAP_sl_rs = THIS.resistance }
			add_to_variable = { ROOT.JAP_sl_rc = 1 }
		}
		set_variable = { JAP_sl_res_inc = 0 }
		if = {
			limit = { check_variable = { JAP_sl_rc > 0 } }
			set_variable = { JAP_sl_res_inc = JAP_sl_rs }
			divide_variable = { JAP_sl_res_inc = JAP_sl_rc }
		}
		# resistance eats the yield: -1% per 1.5% average resistance (75% resistance halves it)
		set_temp_variable = { rk = JAP_sl_res_inc }
		divide_temp_variable = { rk = 150 }
		set_temp_variable = { rkeep = 1 }
		subtract_from_temp_variable = { rkeep = rk }
		clamp_temp_variable = { var = rkeep min = 0.2 max = 1 }
		if = { limit = { check_variable = { JAP_sl_st_inc = 1 } } set_temp_variable = { m = 1.5 } }
		if = { limit = { check_variable = { JAP_sl_st_inc = 2 } } set_temp_variable = { m = 0.8 } }
		if = { limit = { check_variable = { JAP_sl_st_inc = 3 } } set_temp_variable = { m = 0.6 } }

		add_to_temp_variable = { dv = 1 }
		set_variable = { JAP_sl_y_inc = 10 }
		multiply_variable = { JAP_sl_y_inc = m }
		multiply_variable = { JAP_sl_y_inc = dv }
		multiply_variable = { JAP_sl_y_inc = rkeep }
		set_temp_variable = { lost = JAP_sl_loss }
		multiply_temp_variable = { lost = 0.5 }
		set_temp_variable = { keep = 1 }
		subtract_from_temp_variable = { keep = lost }
		multiply_variable = { JAP_sl_y_inc = keep }
		add_to_variable = { JAP_sl_delivered = JAP_sl_y_inc }
		if = { limit = { check_variable = { JAP_sl_st_inc = 1 } } add_to_variable = { JAP_sl_accord = -0.75 } }

	# Chinese states held directly by the army: military administration (lifts under the collaborator government)
	every_state = {
		limit = {
			is_core_of = CHI
			NOT = { is_core_of = MAN }
			JAP_occ_held = yes
			NOT = { controller = { original_tag = RCC } }
			NOT = { has_dynamic_modifier = { modifier = JAP_occ_direct_rule } }
		}
		add_dynamic_modifier = { modifier = JAP_occ_direct_rule }
	}
	every_state = {
		limit = {
			has_dynamic_modifier = { modifier = JAP_occ_direct_rule }
			OR = {
				NOT = { JAP_occ_held = yes }
				controller = { original_tag = RCC }
			}
		}
		remove_dynamic_modifier = { modifier = JAP_occ_direct_rule }
	}
	# every other occupied state: hostile population
	every_state = {
		limit = {
			JAP_occ_held = yes
			NOT = { is_core_of = CHI }
			NOT = { has_dynamic_modifier = { modifier = JAP_occ_hostile_population } }
		}
		add_dynamic_modifier = { modifier = JAP_occ_hostile_population }
	}
	every_state = {

		add_to_variable = { ROOT.JAP_occ_n = 1 }
	}

	multiply_variable = { JAP_occ_growth = 0.006 }
	clamp_variable = { var = JAP_occ_growth min = 0 max = 0.35 }
	set_variable = { JAP_occ_damage = JAP_occ_n }
	multiply_variable = { JAP_occ_damage = 0.005 }
	clamp_variable = { var = JAP_occ_damage min = 0 max = 0.30 }
	if = {
		limit = { NOT = { has_dynamic_modifier = { modifier = JAP_occ_overstretch } } }
		add_dynamic_modifier = { modifier = JAP_occ_overstretch }
	}
	force_update_dynamic_modifier = yes
}
