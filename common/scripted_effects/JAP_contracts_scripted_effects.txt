JAP_cb_update_income = {
	set_variable = { JAP_cb_income = num_of_civilian_factories }
	multiply_variable = { JAP_cb_income = 0.4 }
	if = { limit = { check_variable = { JAP_cb_inf_yasuda > 49 } } multiply_variable = { JAP_cb_income = 1.25 } }
	set_temp_variable = { cb_inflation = JAP_cb_inflation }
	divide_temp_variable = { cb_inflation = -200 }
	add_to_temp_variable = { cb_inflation = 1 }
	multiply_variable = { JAP_cb_income = cb_inflation }
}

JAP_cb_update_perks = {
	if = { limit = { check_variable = { JAP_cb_inf_mitsubishi > 49 } } set_variable = { JAP_zb_industrial_capacity_dockyard = 0.05 } } else = { set_variable = { JAP_zb_industrial_capacity_dockyard = 0 } }
	if = { limit = { check_variable = { JAP_cb_inf_mitsui > 49 } } set_variable = { JAP_zb_local_resources_factor = 0.05 } } else = { set_variable = { JAP_zb_local_resources_factor = 0 } }
	if = { limit = { check_variable = { JAP_cb_inf_sumitomo > 49 } } set_variable = { JAP_zb_production_factory_max_efficiency_factor = 0.03 } } else = { set_variable = { JAP_zb_production_factory_max_efficiency_factor = 0 } }
	if = { limit = { check_variable = { JAP_cb_inf_nissan > 49 } } set_variable = { JAP_zb_production_speed_arms_factory_factor = 0.10 } } else = { set_variable = { JAP_zb_production_speed_arms_factory_factor = 0 } }
	set_temp_variable = { cb_total = 0 }
	add_to_temp_variable = { cb_total = JAP_cb_inf_mitsui }
	add_to_temp_variable = { cb_total = JAP_cb_inf_mitsubishi }
	add_to_temp_variable = { cb_total = JAP_cb_inf_sumitomo }
	add_to_temp_variable = { cb_total = JAP_cb_inf_yasuda }
	add_to_temp_variable = { cb_total = JAP_cb_inf_nissan }
	divide_temp_variable = { cb_total = -1000 }
	set_variable = { JAP_zb_political_power_gain = cb_total }
	force_update_dynamic_modifier = yes
}
