SPR_reopen_mainland_airfields = {
	clr_global_flag = SPR_airfields_closed
	set_global_flag = SPR_airfields_reopened
	every_state = {
		limit = { has_dynamic_modifier = { modifier = SPR_closed_airfields } }
		remove_dynamic_modifier = { modifier = SPR_closed_airfields }
		if = {
			limit = { check_variable = { SPR_stored_air_base > 9 } }
			set_building_level = { type = air_base level = 10 instant_build = yes }
		}
		else_if = {
			limit = { check_variable = { SPR_stored_air_base > 8 } }
			set_building_level = { type = air_base level = 9 instant_build = yes }
		}
		else_if = {
			limit = { check_variable = { SPR_stored_air_base > 7 } }
			set_building_level = { type = air_base level = 8 instant_build = yes }
		}
		else_if = {
			limit = { check_variable = { SPR_stored_air_base > 6 } }
			set_building_level = { type = air_base level = 7 instant_build = yes }
		}
		else_if = {
			limit = { check_variable = { SPR_stored_air_base > 5 } }
			set_building_level = { type = air_base level = 6 instant_build = yes }
		}
		else_if = {
			limit = { check_variable = { SPR_stored_air_base > 4 } }
			set_building_level = { type = air_base level = 5 instant_build = yes }
		}
		else_if = {
			limit = { check_variable = { SPR_stored_air_base > 3 } }
			set_building_level = { type = air_base level = 4 instant_build = yes }
		}
		else_if = {
			limit = { check_variable = { SPR_stored_air_base > 2 } }
			set_building_level = { type = air_base level = 3 instant_build = yes }
		}
		else_if = {
			limit = { check_variable = { SPR_stored_air_base > 1 } }
			set_building_level = { type = air_base level = 2 instant_build = yes }
		}
		else_if = {
			limit = { check_variable = { SPR_stored_air_base > 0 } }
			set_building_level = { type = air_base level = 1 instant_build = yes }
		}
		clear_variable = SPR_stored_air_base
	}
}

# Spanish Civil War: Allied volunteer caps (France and the Soviet Union use their own ideas).
# ENG 3, or 1 after ENG_appeasement. CAN, AST, SAF, NZL, RAJ and POL 1 each.
# An AI minor sends 0 and its slot passes to its human partner: NZL <-> AST, CAN <-> SAF, RAJ -> ENG, POL -> RAJ.
# Runs from on_monthly and when the civil war starts; cleans itself up when scw_over is set.
scw_update_allied_volunteers = {
	if = {
		limit = { OR = { tag = ENG tag = CAN tag = AST tag = SAF tag = NZL tag = RAJ tag = POL } }
		set_temp_variable = { scw_cap = 0 }
		if = {
			limit = {
				has_global_flag = spanish_civil_war
				NOT = { has_global_flag = scw_over }
				NOT = { has_government = fascism }
			}
			if = {
				limit = { tag = ENG }
				set_temp_variable = { scw_cap = 3 }
				if = {
					limit = { has_completed_focus = ENG_appeasement }
					set_temp_variable = { scw_cap = 1 }
				}
				if = {
					limit = { RAJ = { exists = yes is_ai = yes } }
					add_to_temp_variable = { scw_cap = 1 }
				}
				if = {
					limit = { has_idea = ENG_volunteers_idea }
					subtract_from_temp_variable = { scw_cap = 2 }
					if = {
						limit = { check_variable = { scw_cap < 2 } }
						set_temp_variable = { scw_cap = 0 }
					}
				}
			}
			else_if = {
				limit = { is_ai = no }
				set_temp_variable = { scw_cap = 1 }
				if = {
					limit = {
						tag = AST
						NZL = { exists = yes is_ai = yes }
					}
					add_to_temp_variable = { scw_cap = 1 }
				}
				if = {
					limit = {
						tag = NZL
						AST = { exists = yes is_ai = yes }
					}
					add_to_temp_variable = { scw_cap = 1 }
				}
				if = {
					limit = {
						tag = CAN
						SAF = { exists = yes is_ai = yes }
					}
					add_to_temp_variable = { scw_cap = 1 }
				}
				if = {
					limit = {
						tag = SAF
						CAN = { exists = yes is_ai = yes }
					}
					add_to_temp_variable = { scw_cap = 1 }
				}
				if = {
					limit = {
						tag = RAJ
						POL = { exists = yes is_ai = yes }
					}
					add_to_temp_variable = { scw_cap = 1 }
				}
			}
		}
		if = {
			limit = { check_variable = { scw_cap > 0 } }
			set_variable = { scw_volunteer_cap = scw_cap }
			subtract_from_variable = { scw_volunteer_cap = 1 }
			if = {
				limit = { NOT = { has_country_flag = scw_allied_volunteers_flag } }
				set_country_flag = scw_allied_volunteers_flag
				add_relation_rule_override = {
					trigger = scw_allied_volunteer_target
					usage_desc = scw_allied_volunteer_target_tt
					can_send_volunteers = yes
				}
				add_dynamic_modifier = { modifier = scw_allied_volunteers }
			}
		}
		else_if = {
			limit = { has_country_flag = scw_allied_volunteers_flag }
			clr_country_flag = scw_allied_volunteers_flag
			remove_relation_rule_override = {
				trigger = scw_allied_volunteer_target
				can_send_volunteers = yes
			}
			remove_dynamic_modifier = { modifier = scw_allied_volunteers }
			clear_variable = scw_volunteer_cap
		}
	}
}
