# RCC government stages. Called from RCC focuses, RCC events and Japan's China War focuses.

# Dec 1937: the Provisional Government of the Republic of China in Beiping (Wang Kemin).
RCC_become_provisional = {
	set_cosmetic_tag = RCC_provisional
	set_country_flag = RCC_provisional
	add_country_leader_role = {
		character = RCC_wang_kemin
		promote_leader = yes
		country_leader = {
			ideology = despotism
			traits = { RCC_beiyang_financier }
			expire = "1965.1.1.1"
		}
	}
	if = {
		limit = {
			NOT = { 608 = { is_owned_by = ROOT } }
			NOT = { 608 = { is_core_of = MAN } }
			OR = { JAP = { controls_state = 608 } controls_state = 608 }
		}
		transfer_state = 608
	}
	if = {
		limit = {
			NOT = { 614 = { is_owned_by = ROOT } }
			NOT = { 614 = { is_core_of = MAN } }
			OR = { JAP = { controls_state = 614 } controls_state = 614 }
		}
		transfer_state = 614
	}
	if = {
		limit = {
			NOT = { 1039 = { is_owned_by = ROOT } }
			NOT = { 1039 = { is_core_of = MAN } }
			OR = { JAP = { controls_state = 1039 } controls_state = 1039 }
		}
		transfer_state = 1039
	}
	if = {
		limit = {
			NOT = { 597 = { is_owned_by = ROOT } }
			NOT = { 597 = { is_core_of = MAN } }
			OR = { JAP = { controls_state = 597 } controls_state = 597 }
		}
		transfer_state = 597
	}
	if = {
		limit = {
			NOT = { 1038 = { is_owned_by = ROOT } }
			NOT = { 1038 = { is_core_of = MAN } }
			OR = { JAP = { controls_state = 1038 } controls_state = 1038 }
		}
		transfer_state = 1038
	}

		limit = { owns_state = 608 }
		set_capital = { state = 608 }
	}
}

# Mar 1940: Wang Jingwei returns the capital to Nanjing as the Reorganized National Government.
RCC_become_reorganized = {
	set_cosmetic_tag = RCC_reorganized
	set_country_flag = RCC_reorganized
	add_country_leader_role = {
		character = RCC_wang_jingwei
		promote_leader = yes
		country_leader = {
			ideology = despotism
			traits = { RCC_heir_of_sun_yat_sen RCC_bullet_of_1935_1 }
			expire = "1965.1.1.1"
		}
	}
	if = {
		limit = { owns_state = 1035 }
		set_capital = { state = 1035 }
		1035 = { add_core_of = ROOT }
	}
}
