# Targeted variant of PACIFIC_state_swap: updates only the state in scope.
# Used from on_state_control_changed so a European front-line flip during
# Barbarossa does not re-sweep the whole 95-state array on every change.
PACIFIC_state_swap_single = {
	# If Controlled by the allies make it the allied modifier
	if = {
		limit = {
			OR = {
				THIS.controller = { is_in_faction_with = ENG }
				THIS.controller = { is_in_faction_with = USA }
				THIS.controller = { is_in_faction_with = FRA }
				THIS = {
					OR = {
						is_controlled_by = ENG
						is_controlled_by = USA
						is_controlled_by = FRA
						is_controlled_by = INS
					}
				}
			}
		}
		if = {
			limit = {
				NOT = {	has_dynamic_modifier = { modifier = PACIFIC_COMMONWEALTH_allied_controlled } }
			}
			add_dynamic_modifier = { modifier = PACIFIC_COMMONWEALTH_allied_controlled }
		}
		if = {
			limit = {
				has_dynamic_modifier = { modifier = PACIFIC_COMMONWEALTH_axis_controlled }
			}
			remove_dynamic_modifier = { modifier = PACIFIC_COMMONWEALTH_axis_controlled }
		}
	}
	# If Controlled by the axis make it the axis modifier
	else_if = {
		limit = {
			OR = {
				THIS.controller = { is_in_faction_with = GER }
				THIS.controller = { is_in_faction_with = JAP }
				THIS.controller = { is_in_faction_with = ITA }
				THIS = {
					OR = {
						is_controlled_by = GER
						is_controlled_by = JAP
						is_controlled_by = ITA
						is_controlled_by = FRA.SOUTHFRANCEOWNER
					}
				}
			}
		}
		if = {
			limit = {
				NOT = {	has_dynamic_modifier = { modifier = PACIFIC_COMMONWEALTH_axis_controlled } }
			}
			add_dynamic_modifier = { modifier = PACIFIC_COMMONWEALTH_axis_controlled }
		}
		if = {
			limit = {
				has_dynamic_modifier = { modifier = PACIFIC_COMMONWEALTH_allied_controlled }
			}
			remove_dynamic_modifier = { modifier = PACIFIC_COMMONWEALTH_allied_controlled }
		}
	}
}

### State of Africa ###
AFRICA_recalculate_penalty = {
	set_variable = { global.AFRICA_DEBUFF_INVASION_PENALTY = global.AFRICA_BASES_AXIS }
	multiply_variable = { global.AFRICA_DEBUFF_INVASION_PENALTY = 0.45 }
	divide_variable = { global.AFRICA_DEBUFF_INVASION_PENALTY = 9 }
	if = {
		limit = { check_variable = { global.AFRICA_SOV_DECAY > global.AFRICA_DEBUFF_INVASION_PENALTY } }
		set_variable = { global.AFRICA_SOV_DECAY = global.AFRICA_DEBUFF_INVASION_PENALTY }
	}
	subtract_from_variable = { global.AFRICA_DEBUFF_INVASION_PENALTY = global.AFRICA_SOV_DECAY }
	if = {
		limit = { check_variable = { global.AFRICA_BASES_AXIS < 1 } }
		set_variable = { global.AFRICA_DEBUFF_INVASION_PENALTY = -0.1 }
	}
	set_variable = { global.SOTA_ALLIES_LOCATIONS = 9 }
	subtract_from_variable = { global.SOTA_ALLIES_LOCATIONS = global.AFRICA_BASES_AXIS }
}

# sov
AFRICA_count_sov_cities = {
	set_variable = { global.AFRICA_SOV_CITIES_HELD = 0 }
	if = { limit = { 195 = { is_controlled_by = SOV } } add_to_variable = { global.AFRICA_SOV_CITIES_HELD = 1 } }
	if = { limit = { 219 = { is_controlled_by = SOV } } add_to_variable = { global.AFRICA_SOV_CITIES_HELD = 1 } }
	if = { limit = { 217 = { is_controlled_by = SOV } } add_to_variable = { global.AFRICA_SOV_CITIES_HELD = 1 } }
	set_variable = { global.AFRICA_SOV_CITIES_LOST = 3 }
	subtract_from_variable = { global.AFRICA_SOV_CITIES_LOST = global.AFRICA_SOV_CITIES_HELD }
}

AFRICA_sov_cities_tick = {
	if = {
		limit = {
			count_triggers = {
				amount = 2
				195 = { NOT = { is_controlled_by = SOV } }
				219 = { NOT = { is_controlled_by = SOV } }
				217 = { NOT = { is_controlled_by = SOV } }
			}
		}
		add_to_variable = { global.AFRICA_SOV_DECAY = 0.025 }
	}
	else_if = {
		limit = { check_variable = { global.AFRICA_SOV_DECAY > 0 } }
		subtract_from_variable = { global.AFRICA_SOV_DECAY = 0.025 }
		if = {
			limit = { check_variable = { global.AFRICA_SOV_DECAY < 0 } }
			set_variable = { global.AFRICA_SOV_DECAY = 0 }
		}
	}
	AFRICA_count_sov_cities = yes
	AFRICA_recalculate_penalty = yes
}

# Recount
AFRICA_recount_bases = {
	set_variable = { global.AFRICA_BASES_AXIS = 0 }
	every_state = {
		limit = {
			OR = {
				state = 461 state = 459 state = 458
				state = 448 state = 450 state = 447
				state = 454 state = 551 state = 291
			}
		}
		if = {
			limit = { AFRICA_is_axis_controlled = yes }
			set_state_flag = NOT_AFRICA_ALLIES_controls
			clr_state_flag = AFRICA_ALLIES_controls
			add_to_variable = { global.AFRICA_BASES_AXIS = 1 }
		}
		else = {
			set_state_flag = AFRICA_ALLIES_controls
			clr_state_flag = NOT_AFRICA_ALLIES_controls
		}
	}
	AFRICA_recalculate_penalty = yes
}

AFRICA_state_swap_single = {
	if = {
		limit = { AFRICA_is_allied_controlled = yes }
		if = {
			limit = {
				has_dynamic_modifier = { modifier = AFRICA_axis_controlled }
			}
			remove_dynamic_modifier = { modifier = AFRICA_axis_controlled }
		}
	}
	else_if = {
		limit = { AFRICA_is_axis_controlled = yes }
		if = {
			limit = {
				NOT = {	has_dynamic_modifier = { modifier = AFRICA_axis_controlled } }
			}
			add_dynamic_modifier = { modifier = AFRICA_axis_controlled }
		}
	}
}
