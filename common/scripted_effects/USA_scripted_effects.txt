# ── Third New Deal — single variable-driven spirit ─────────────────────────
# Base values are set on grant; acts/paths/events apply deltas to the
# USA_tnd_* variables and force a refresh so the changes display live.



# Removes the Third New Deal spirit and clears its variables.
USA_remove_all_third_new_deal = {
	if = {
		limit = { has_dynamic_modifier = { modifier = USA_third_new_deal } }
		remove_dynamic_modifier = { modifier = USA_third_new_deal }
	}
	clear_variable = USA_tnd_infra
	clear_variable = USA_tnd_ic
	clear_variable = USA_tnd_ppcost
	clear_variable = USA_tnd_consumer
	clear_variable = USA_tnd_factory
}


USA_remove_new_business_deal = {

# Eases the Great Depression by one severity level. Applies the level-specific
# modifier deltas via add_to_variable (each shows its own native modifier line in
# the tooltip) and refreshes the var-driven USA_great_depression dynamic modifier.
USA_ease_depression = {
	if = {
		limit = { check_variable = { var = USA_depression_level value = 3.5 compare = greater_than_or_equals } } # Full Crisis -> Severe
		add_to_variable = { var = USA_depression_cg           value = -0.05  tooltip = consumer_goods_expected_value_tt }
		add_to_variable = { var = USA_depression_resources    value = 0.10   tooltip = local_resources_factor_tt }
		add_to_variable = { var = USA_depression_conscription value = 0.005  tooltip = conscription_tt }
		add_to_variable = { var = USA_depression_construction value = 0.05   tooltip = production_speed_buildings_factor_tt }
		set_variable = { var = USA_depression_level value = 3 }
		force_update_dynamic_modifier = yes
	}
	else_if = {
		limit = { check_variable = { var = USA_depression_level value = 2.5 compare = greater_than_or_equals } } # Severe -> Moderate
		add_to_variable = { var = USA_depression_cg           value = -0.05  tooltip = consumer_goods_expected_value_tt }
		add_to_variable = { var = USA_depression_resources    value = 0.10   tooltip = local_resources_factor_tt }
		add_to_variable = { var = USA_depression_conscription value = 0.003  tooltip = conscription_tt }
		add_to_variable = { var = USA_depression_construction value = 0.03   tooltip = production_speed_buildings_factor_tt }
		set_variable = { var = USA_depression_level value = 2 }
		force_update_dynamic_modifier = yes
	}
	else_if = {
		limit = { check_variable = { var = USA_depression_level value = 1.5 compare = greater_than_or_equals } } # Moderate -> Mild
		add_to_variable = { var = USA_depression_cg           value = -0.05  tooltip = consumer_goods_expected_value_tt }
		add_to_variable = { var = USA_depression_resources    value = 0.10   tooltip = local_resources_factor_tt }
		add_to_variable = { var = USA_depression_conscription value = 0.004  tooltip = conscription_tt }
		add_to_variable = { var = USA_depression_construction value = 0.04   tooltip = production_speed_buildings_factor_tt }
		set_variable = { var = USA_depression_level value = 1 }
		force_update_dynamic_modifier = yes
	}
	else_if = {
		limit = { check_variable = { var = USA_depression_level value = 0.5 compare = greater_than_or_equals } } # Mild -> Recovered
		add_to_variable = { var = USA_depression_cg           value = -0.05  tooltip = consumer_goods_expected_value_tt }
		add_to_variable = { var = USA_depression_resources    value = 0.10   tooltip = local_resources_factor_tt }
		add_to_variable = { var = USA_depression_conscription value = 0.003  tooltip = conscription_tt }
		add_to_variable = { var = USA_depression_construction value = 0.03   tooltip = production_speed_buildings_factor_tt }
		set_variable = { var = USA_depression_level value = 0 }
		force_update_dynamic_modifier = yes
	}
}


# Cash and Carry
# Requires temp variable cnc_invest. The decision's own cost has already taken the PP.
USA_cnc_invest_pool = {
	if = {
		limit = { has_global_flag = USA_cnc_eximbank }
		multiply_temp_variable = { cnc_invest = 1.25 }
	}
	add_to_variable = {
		global.cnc_pp_pool = cnc_invest
		tooltip = USA_cnc_add_to_pool_tt
	}
	add_to_variable = { cnc_pp_invested = cnc_invest }
}

# Cash and Carry — pay for a package from the pool (cost set by the chosen package size).
USA_cnc_pay_from_pool = {
	subtract_from_variable = { global.cnc_pp_pool = cnc_pkg_cost }
	add_to_variable = { cnc_pp_drawn = cnc_pkg_cost }
	# Every 5 lots sold gives the USA +2.5% factory output (War Economy of the United States), no cap.
	custom_effect_tooltip = USA_cnc_sale_output_tt
	hidden_effect = {
		set_temp_variable = { cnc_sold = cnc_pkg_lots }
		add_to_variable = { global.cnc_lots_sold = cnc_sold }
		USA = {
			add_to_variable = { USA_cnc_lots_progress = cnc_sold }
			while_loop_effect = {
				limit = { check_variable = { USA_cnc_lots_progress > 4.5 } }
				subtract_from_variable = { USA_cnc_lots_progress = 5 }
				add_to_variable = { USA_cnc_output_var = 0.025 }	# market overview counter
				add_to_variable = { USA_we_output = 0.025 }		# War Economy of the United States
			}
		}
	}
}

# 1936 campaign minigame: push the polling bar toward the candidate the USA is backing
# (USA_campaign_side: -1 = Roosevelt, +1 = Landon). Pushback goes the other way.
# Every swing is a multiple of 0.04 (as are the +/-0.12 swings of the campaign events) and the bar starts
# at -0.10, so it stays on values that are 2 mod 4 in hundredths: it can never land on exactly 0.
USA_campaign_push_4 = {
	if = {
		limit = { check_variable = { USA_campaign_side < 0 } }
		add_power_balance_value = { id = USA_1936_election_bop value = -0.04 }
	}
	else = {
		add_power_balance_value = { id = USA_1936_election_bop value = 0.04 }
	}
}
