# ── Third New Deal — single variable-driven spirit ─────────────────────────
# Base values are set on grant; acts/paths/events apply deltas to the
# USA_tnd_* variables and force a refresh so the changes display live.

# Grants the Third New Deal spirit and initialises its base values.
USA_grant_third_new_deal = {
	hidden_effect = {
		set_variable = { USA_tnd_infra     = 0 }
		set_variable = { USA_tnd_ic        = 0 }
		set_variable = { USA_tnd_ppcost    = 0 }
		set_variable = { USA_tnd_consumer  = 0 }
		set_variable = { USA_tnd_factory   = 0 }
	}
	custom_effect_tooltip = USA_add_third_new_deal_tt
	add_stability = 0.10
	add_to_variable = { var = USA_tnd_infra     value = 0.50 tooltip = production_speed_infrastructure_factor_tt }
	add_to_variable = { var = USA_tnd_ic        value = 0.40 tooltip = production_speed_industrial_complex_factor_tt }
	add_to_variable = { var = USA_tnd_ppcost    value = 0.35 tooltip = political_power_cost_tt }
	hidden_effect = { add_dynamic_modifier = { modifier = USA_third_new_deal } }
	force_update_dynamic_modifier = yes
}

# Removes the Third New Deal spirit and clears its variables.
USA_remove_all_third_new_deal = {
	if = {
		limit = { has_dynamic_modifier = { modifier = USA_third_new_deal } }
		remove_dynamic_modifier = { modifier = USA_third_new_deal }
	}
	clear_variable = USA_tnd_infra
	clear_variable = USA_tnd_ic
	clear_variable = USA_tnd_ppcost
	clear_variable = USA_tnd_consumer
	clear_variable = USA_tnd_factory
}

# Eases the Great Depression by one severity level. Applies the level-specific
# modifier deltas via add_to_variable (each shows its own native modifier line in
# the tooltip) and refreshes the var-driven USA_great_depression dynamic modifier.
USA_ease_depression = {
	if = {
		limit = { check_variable = { var = USA_depression_level value = 3.5 compare = greater_than_or_equals } } # Full Crisis -> Severe
		add_to_variable = { var = USA_depression_cg           value = -0.05  tooltip = consumer_goods_expected_value_tt }
		add_to_variable = { var = USA_depression_resources    value = 0.10   tooltip = local_resources_factor_tt }
		add_to_variable = { var = USA_depression_conscription value = 0.005  tooltip = conscription_tt }
		add_to_variable = { var = USA_depression_construction value = 0.05   tooltip = production_speed_buildings_factor_tt }
		set_variable = { var = USA_depression_level value = 3 }
		force_update_dynamic_modifier = yes
	}
	else_if = {
		limit = { check_variable = { var = USA_depression_level value = 2.5 compare = greater_than_or_equals } } # Severe -> Moderate
		add_to_variable = { var = USA_depression_cg           value = -0.05  tooltip = consumer_goods_expected_value_tt }
		add_to_variable = { var = USA_depression_resources    value = 0.10   tooltip = local_resources_factor_tt }
		add_to_variable = { var = USA_depression_conscription value = 0.003  tooltip = conscription_tt }
		add_to_variable = { var = USA_depression_construction value = 0.03   tooltip = production_speed_buildings_factor_tt }
		set_variable = { var = USA_depression_level value = 2 }
		force_update_dynamic_modifier = yes
	}
	else_if = {
		limit = { check_variable = { var = USA_depression_level value = 1.5 compare = greater_than_or_equals } } # Moderate -> Mild
		add_to_variable = { var = USA_depression_cg           value = -0.05  tooltip = consumer_goods_expected_value_tt }
		add_to_variable = { var = USA_depression_resources    value = 0.10   tooltip = local_resources_factor_tt }
		add_to_variable = { var = USA_depression_conscription value = 0.004  tooltip = conscription_tt }
		add_to_variable = { var = USA_depression_construction value = 0.04   tooltip = production_speed_buildings_factor_tt }
		set_variable = { var = USA_depression_level value = 1 }
		force_update_dynamic_modifier = yes
	}
	else_if = {
		limit = { check_variable = { var = USA_depression_level value = 0.5 compare = greater_than_or_equals } } # Mild -> Recovered
		add_to_variable = { var = USA_depression_cg           value = -0.05  tooltip = consumer_goods_expected_value_tt }
		add_to_variable = { var = USA_depression_resources    value = 0.10   tooltip = local_resources_factor_tt }
		add_to_variable = { var = USA_depression_conscription value = 0.003  tooltip = conscription_tt }
		add_to_variable = { var = USA_depression_construction value = 0.03   tooltip = production_speed_buildings_factor_tt }
		set_variable = { var = USA_depression_level value = 0 }
		force_update_dynamic_modifier = yes
	}
}


# Cash and Carry
# Requires temp variable cnc_invest. The decision's own cost has already taken the PP.
USA_cnc_invest_pool = {
	if = {
		limit = { has_global_flag = USA_cnc_eximbank }
		multiply_temp_variable = { cnc_invest = 1.25 }
	}
	add_to_variable = {
		global.cnc_pp_pool = cnc_invest
		tooltip = USA_cnc_add_to_pool_tt
	}
	add_to_variable = { cnc_pp_invested = cnc_invest }
}

# Cash and Carry — pay for a package from the pool (cost set by the chosen package size).
USA_cnc_pay_from_pool = {
	subtract_from_variable = { global.cnc_pp_pool = cnc_pkg_cost }
	add_to_variable = { cnc_pp_drawn = cnc_pkg_cost }
	# Every 5 lots sold gives the USA +2.5% factory output (War Economy of the United States), no cap.
	custom_effect_tooltip = USA_cnc_sale_output_tt
	hidden_effect = {
		set_temp_variable = { cnc_sold = cnc_pkg_lots }
		add_to_variable = { global.cnc_lots_sold = cnc_sold }
		USA = {
			add_to_variable = { USA_cnc_lots_progress = cnc_sold }
			while_loop_effect = {
				limit = { check_variable = { USA_cnc_lots_progress > 4.5 } }
				subtract_from_variable = { USA_cnc_lots_progress = 5 }
				add_to_variable = { USA_cnc_output_var = 0.025 }
			}
			USA_recalc_war_economy = yes
		}
	}
}

# War Economy of the United States (USA_war_economy dynamic modifier). Call in the USA scope whenever
# the market output, the December 7th war economy or Big Business Subsidies change.
#   flags: USA_war_economy_active (December 7th done), USA_war_economy_landon_path (Willkie path),
#          USA_war_economy_subsidies_on (Offer Subsidies to Big Business done)
USA_recalc_war_economy = {
	set_variable = { USA_we_output = USA_cnc_output_var }
	set_variable = { USA_we_arms_speed = 0 }
	set_variable = { USA_we_dock_speed = 0 }
	set_variable = { USA_we_arms_cost = 0 }
	set_variable = { USA_we_consumer = 0 }
	set_variable = { USA_we_slots = 0 }
	set_variable = { USA_we_max_eff = 0 }
	set_variable = { USA_we_eff_gain = 0 }
	set_variable = { USA_we_start_eff = 0 }
	set_variable = { USA_we_cg_expected = 0 }
	set_variable = { USA_we_pp_gain = 0 }
	set_variable = { USA_we_conscription = 0 }
	set_variable = { USA_we_stability = 0 }
	set_variable = { USA_we_war_support = 0 }
	set_variable = { USA_we_stab_factor = 0 }
	set_variable = { USA_we_ws_factor = 0 }
	if = {
		limit = { has_country_flag = USA_war_economy_active }
		set_variable = { USA_we_consumer = -0.10 }
		set_variable = { USA_we_slots = 2 }
		if = {
			limit = { has_country_flag = USA_war_economy_landon_path }
			add_to_variable = { USA_we_output = 0.45 }
			set_variable = { USA_we_arms_speed = 0.50 }
			set_variable = { USA_we_dock_speed = 0.40 }
			set_variable = { USA_we_arms_cost = -0.15 }
		}
		else = {
			add_to_variable = { USA_we_output = 0.35 }
			set_variable = { USA_we_arms_speed = 0.70 }
			set_variable = { USA_we_dock_speed = 0.35 }
		}
		if = {
			limit = { has_country_flag = USA_war_economy_subsidies_on }
			set_variable = { USA_we_max_eff = 0.10 }
			add_to_variable = { USA_we_eff_gain = 0.05 }
		}
	}
	# Wartime agencies (flags set by their focuses)
	if = {
		limit = { has_country_flag = USA_we_bonds }                # Series E War Bonds (Issue War Bonds)
		add_to_variable = { USA_we_cg_expected = -0.025 }
	}
	if = {
		limit = { has_country_flag = USA_we_wpa }                  # War Powers Act
		add_to_variable = { USA_we_pp_gain = 0.15 }
		add_to_variable = { USA_we_conscription = 0.05 }
		add_to_variable = { USA_we_stability = -0.001 }
	}
	if = {
		limit = { has_country_flag = USA_we_owm }                  # Office of War Mobilisation
		add_to_variable = { USA_we_eff_gain = 0.10 }
		add_to_variable = { USA_we_start_eff = 0.10 }
	}
	if = {
		limit = { has_country_flag = USA_we_owi }                  # Office of War Information
		add_to_variable = { USA_we_war_support = 0.003 }
		add_to_variable = { USA_we_stability = 0.002 }
	}
	if = {
		limit = { has_country_flag = USA_we_epca }                 # Emergency Price Control Act
		add_to_variable = { USA_we_cg_expected = -0.05 }
		add_to_variable = { USA_we_eff_gain = 0.05 }
		add_to_variable = { USA_we_stability = 0.001 }
	}
	# 1936 election mandate (flags set for 730 days by event usa.200, by margin of victory)
	if = {
		limit = { has_country_flag = USA_we_mandate_fdr_1 }
		add_to_variable = { USA_we_stab_factor = 0.03 }
		add_to_variable = { USA_we_pp_gain = 0.05 }
	}
	if = {
		limit = { has_country_flag = USA_we_mandate_fdr_2 }
		add_to_variable = { USA_we_stab_factor = 0.05 }
		add_to_variable = { USA_we_pp_gain = 0.1 }
	}
	if = {
		limit = { has_country_flag = USA_we_mandate_fdr_3 }
		add_to_variable = { USA_we_stab_factor = 0.08 }
		add_to_variable = { USA_we_pp_gain = 0.15 }
		add_to_variable = { USA_we_ws_factor = 0.03 }
	}
	if = {
		limit = { has_country_flag = USA_we_mandate_lan_1 }
		add_to_variable = { USA_we_stab_factor = 0.03 }
		add_to_variable = { USA_we_output = 0.02 }
	}
	if = {
		limit = { has_country_flag = USA_we_mandate_lan_2 }
		add_to_variable = { USA_we_stab_factor = 0.05 }
		add_to_variable = { USA_we_output = 0.04 }
	}
	if = {
		limit = { has_country_flag = USA_we_mandate_lan_3 }
		add_to_variable = { USA_we_stab_factor = 0.08 }
		add_to_variable = { USA_we_ws_factor = 0.03 }
		add_to_variable = { USA_we_output = 0.06 }
	}
	force_update_dynamic_modifier = yes
}
