		limit = {
			SOV = {
				exists = yes
				has_war_with = JAP
			}
		}

		set_temp_variable = { JAP_soviet_war = 1 }
		set_variable = { JAP_fourth_occupation_recipient = SOV }
	}
	else = {
		every_country = {
			limit = {
				exists = yes
				has_war_with = JAP
				NOT = {
					
					original_tag = ENG
					original_tag = FRA
					original_tag = USA
					
				}
			}

			if = {
				limit = {
					check_variable = {
						var = JAP_candidate_war_score
						value = ROOT.JAP_highest_eligible_war_score
						compare = greater_than
					}
				}

				ROOT = {
					set_temp_variable = {
						JAP_highest_eligible_war_score = PREV.JAP_candidate_war_score
					}
