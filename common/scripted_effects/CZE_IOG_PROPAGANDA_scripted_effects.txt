
	CZE_IOG_pick_propaganda_campaign_slot_effect = {
		hidden_effect = {
			if = {
				limit = {
					check_variable = { CZE_IOG_poster_left_icon_frame = CZE_IOG_propaganda_empty_slot_index }
				}
				set_variable = { CZE_IOG_poster_left_icon_frame = CZE_IOG_new_propaganda_campaign }

				set_variable = { CZE_IOG_propaganda_left_poster_to_age = CZE_IOG_poster_left_icon_frame }
			}
			else_if = {
				limit = {
					check_variable = { CZE_IOG_poster_center_icon_frame = CZE_IOG_propaganda_empty_slot_index }
				}
				set_variable = { CZE_IOG_poster_center_icon_frame = CZE_IOG_new_propaganda_campaign }

				set_variable = { CZE_IOG_propaganda_center_poster_to_age = CZE_IOG_poster_center_icon_frame }
			}
			set_variable = { CZE_IOG_new_propaganda_campaign = 0 }
			add_to_variable = { CZE_IOG_propaganda_campaigns_taken = 1 }
		}
	}

	CZE_IOG_reset_propaganda_campaign_slot_effect = {
		hidden_effect = {
			if = {
				limit = {
					check_variable = { CZE_IOG_poster_left_icon_frame = CZE_IOG_propaganda_campaign_to_remove }
				}
				set_variable = { CZE_IOG_poster_left_icon_frame = CZE_IOG_propaganda_empty_slot_index }

				clear_variable = CZE_IOG_propaganda_left_poster_to_age
				clr_country_flag = CZE_IOG_propaganda_left_slot_is_old_flag
			}
			else_if = {
				limit = {
					check_variable = { CZE_IOG_poster_center_icon_frame = CZE_IOG_propaganda_campaign_to_remove }
				}
				set_variable = { CZE_IOG_poster_center_icon_frame = CZE_IOG_propaganda_empty_slot_index }

				clear_variable = CZE_IOG_propaganda_center_poster_to_age
				clr_country_flag = CZE_IOG_propaganda_center_slot_is_old_flag
			}
			set_variable = { CZE_IOG_propaganda_campaign_to_remove = 0 }
		}
	}

	CZE_IOG_unlock_propaganda_campaign_slot_effect = {
		hidden_effect = {
			if = {
				limit = {
					check_variable = { CZE_IOG_poster_left_icon_frame = CZE_IOG_propaganda_locked_slot_index }
				}
				set_variable = { CZE_IOG_poster_left_icon_frame = CZE_IOG_propaganda_empty_slot_index }
			}
			else_if = {
				limit = {
					check_variable = { CZE_IOG_poster_center_icon_frame = CZE_IOG_propaganda_locked_slot_index }
				}
				set_variable = { CZE_IOG_poster_center_icon_frame = CZE_IOG_propaganda_empty_slot_index }
			}
		}
	}
