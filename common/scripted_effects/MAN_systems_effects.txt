		limit = { check_variable = { MAN_h_mgy < 0.5 } }
		set_variable = { MAN_h_mgy = 1 }
		set_variable = { MAN_fav_mgy = 45 }
		custom_effect_tooltip = MAN_tt_ayukawa_arrives
	}
}
