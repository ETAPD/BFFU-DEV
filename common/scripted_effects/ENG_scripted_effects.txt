# Grants ENG_colonial_development_boost (+20% civ construction, +15% infra
# construction) to each Dominion/Colony while the UK is actively developing it,
# and strips it once development for that target ends.Run weekly from on_weekly_ENG; ROOT =
# ENG. add_idea 
ENG_sync_colonial_development_boost = {
	if = {
		limit = { AST = { exists = yes } }
		if = {
			limit = {
				has_completed_focus = ENG_develop_ast
				NOT = { has_completed_focus = ENG_finish_development }
				AST = { NOT = { all_core_state = { has_state_flag = ENG_development_marker } } }
			}
			AST = { add_ideas = ENG_colonial_development_boost }
			else = { AST = { remove_ideas = ENG_colonial_development_boost } }
		}
	}
	if = {
		limit = { NZL = { exists = yes } }
		if = {
			limit = {
				has_completed_focus = ENG_develop_nzl
				NOT = { has_completed_focus = ENG_finish_development }
				NZL = { NOT = { all_core_state = { has_state_flag = ENG_development_marker } } }
			}
			NZL = { add_ideas = ENG_colonial_development_boost }
			else = { NZL = { remove_ideas = ENG_colonial_development_boost } }
		}
	}
	if = {
		limit = { CAN = { exists = yes } }
		if = {
			limit = {
				has_completed_focus = ENG_develop_can
				NOT = { has_completed_focus = ENG_finish_development }
				CAN = { NOT = { all_core_state = { has_state_flag = ENG_development_marker } } }
			}
			CAN = { add_ideas = ENG_colonial_development_boost }
			else = { CAN = { remove_ideas = ENG_colonial_development_boost } }
		}
	}
	if = {
		limit = { SAF = { exists = yes } }
		if = {
			limit = {
				has_completed_focus = ENG_develop_saf
				NOT = { has_completed_focus = ENG_finish_development }
				SAF = { NOT = { all_core_state = { has_state_flag = ENG_development_marker } } }
			}
			SAF = { add_ideas = ENG_colonial_development_boost }
			else = { SAF = { remove_ideas = ENG_colonial_development_boost } }
		}
	}
	if = {
		limit = { RAJ = { exists = yes } }
		if = {
			limit = {
				has_completed_focus = ENG_develop_raj
				NOT = { has_completed_focus = ENG_finish_development }
				RAJ = { NOT = { all_core_state = { has_state_flag = ENG_development_marker } } }
			}
			RAJ = { add_ideas = ENG_colonial_development_boost }
			else = { RAJ = { remove_ideas = ENG_colonial_development_boost } }
		}
	}
}

ENG_shadow_debug_calc = {
	set_variable = { var = shadow_dbg_current value = 0 }
	for_each_scope_loop = {
		array = ENG.starting_core_states
		set_temp_variable = { var = tmp_civs value = industrial_complex_level }
		ROOT = {
			add_to_variable = { var = shadow_dbg_current value = temp:tmp_civs }
		}
	}
	set_variable = { var = shadow_dbg_delta value = var:shadow_dbg_current }
	subtract_from_variable = { var = shadow_dbg_delta value = var:shadow_civs1 }
	clamp_variable = { var = shadow_dbg_delta min = 0 max = 120 }
	set_variable = { var = shadow_dbg_mils value = var:shadow_dbg_delta }
	divide_variable = { var = shadow_dbg_mils value = 3 }
	set_variable = { var = shadow_dbg_tmp value = var:shadow_dbg_mils }
	round_variable = shadow_dbg_mils
	if = {
		limit = {
			check_variable = {
				var = shadow_dbg_tmp
				value = var:shadow_dbg_mils
				compare = less_than
			}
		}
		subtract_from_variable = { var = shadow_dbg_mils value = 1 }
	}
}
