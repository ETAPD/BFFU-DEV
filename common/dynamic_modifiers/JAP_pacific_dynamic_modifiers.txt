# Festung Europa - shows how the Nanshin-ron Pacific war limits the Allies' ability to land in Europe.
#
# JAP_festung_europa is a STATE modifier placed on every European coastal state. It is DISPLAY ONLY
# (custom_modifier_tooltip), because the engine's naval_invasion_penalty is a country-scope modifier
# and does nothing on a state. The real mechanical effect lives on the Allied invaders below.
#
# JAP_festung_allied_diversion is a COUNTRY modifier added to the Allied majors (USA, ENG).
# Globals are recomputed in JAP_pacific_recompute. See [[JAP_pacific_scripted_effects]].

JAP_festung_europa = {
	enable = { always = yes }
	icon = GFX_idea_jap_militarism_shogun_small
	custom_modifier_tooltip = JAP_festung_europa_tt
}

JAP_festung_allied_diversion = {
	enable = { always = yes }
	icon = GFX_idea_jap_militarism_shogun_small
	naval_invasion_penalty = global.JAP_festung_inv    # +per island: harder Allied landings
}
