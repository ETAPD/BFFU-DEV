# War Economy of the United States 
# Every value is driven by a variable that USA_recalc_war_economy rewrites (USA_scripted_effects.txt):
#   - day one: only the Cash and Carry market output (-10% when the Neutrality Act opens the market,
#     +2.5% per 5 lots sold)
#   - December 7th adds the war economy (Roosevelt or Willkie path), Big Business Subsidies add efficiency.
#   - the wartime agencies (Series E War Bonds, War Powers Act, Office of War Mobilisation, Office of War
#     Information, Emergency Price Control Act) add to it instead of being separate national spirits.
#   - the 1936 election mandate (Roosevelt or Landon, by margin of victory) is added to it for two years.
USA_war_economy = {
	icon = GFX_idea_war_economy
	enable = { tag = USA }

	industrial_capacity_factory               = USA_we_output
	production_speed_arms_factory_factor      = USA_we_arms_speed
	production_speed_dockyard_factor          = USA_we_dock_speed
	production_cost_arms_factory_factor       = USA_we_arms_cost
	consumer_goods_factor                     = USA_we_consumer
	global_building_slots_factor              = USA_we_slots
	production_factory_max_efficiency_factor  = USA_we_max_eff
	production_factory_efficiency_gain_factor = USA_we_eff_gain
	production_factory_start_efficiency_factor = USA_we_start_eff
	consumer_goods_expected_value             = USA_we_cg_expected
	political_power_gain                      = USA_we_pp_gain
	conscription_factor                       = USA_we_conscription
	stability_weekly                          = USA_we_stability
	war_support_weekly                        = USA_we_war_support
	stability_factor                          = USA_we_stab_factor
	war_support_factor                        = USA_we_ws_factor
}
