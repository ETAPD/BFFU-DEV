# history/countries/SAF - South Africa.txt.

#                               whole game. Opens deep in the Depression
#   SAF_the_afrikaner_question - permanent. The Union lives with it.
#   SAF_union_defence_force   - permanent. Only the military branches
#                               (army/air/navy) deposit here.
###########################################################################

### the industry branch stays throttled until this is addressed.

	enable = { always = yes }

	icon = GFX_idea_saf_history_of_segregation
