MAN_plan_cat = {
	MAN_site_gongchangling = {
		icon = GFX_decision_steel
		priority = 170
		allowed = { original_tag = MAN }
		visible = { has_country_flag = MAN_plan_unlocked  }
		state_target = yes
		targets = { 716 }
		target_trigger = {
			FROM = {
				is_owned_and_controlled_by = ROOT
				has_state_flag = MAN_surveyed
				has_state_flag = MAN_zone_min
				OR = {
					NOT = { check_variable = { MAN_lv_gongchangling > 1.5 } }
					AND = {
						ROOT = { has_country_flag = MAN_second_five_year_plan_active }
						NOT = { check_variable = { MAN_lv_gongchangling > 2.5 } }
					}
				}
			}
		}
		on_map_mode = map_and_decisions_view
		available = {
			custom_trigger_tooltip = { tooltip = MAN_trg_not_busy FROM = { NOT = { has_state_flag = MAN_busy } } }
			custom_trigger_tooltip = { tooltip = MAN_trg_slot check_variable = { MAN_proj_active < MAN_proj_slots } }
		}
		custom_cost_trigger = { has_political_power > 19.99 check_variable = { MAN_labour > 14.99 } }
		custom_cost_text = MAN_cc_20_15_0
		fire_only_once = no
		days_remove = 90
		modifier = { civilian_factory_use = 1 }
		cancel_trigger = { FROM = { NOT = { is_owned_and_controlled_by = ROOT } } }
		cancel_effect = {
			subtract_from_variable = { MAN_proj_active = 1 }
			FROM = { clr_state_flag = MAN_busy }
		}
		complete_effect = {
			add_political_power = -20
			subtract_from_variable = { MAN_labour = 15 }
			add_to_variable = { MAN_proj_active = 1 }
			FROM = { set_state_flag = MAN_busy }
			hidden_effect = { MAN_ui_refresh = yes }
		}
		remove_effect = {
			subtract_from_variable = { MAN_proj_active = 1 }
			add_to_variable = { MAN_proj_done = 1 }
			FROM = {
				clr_state_flag = MAN_busy
				add_to_variable = { MAN_lv_gongchangling = 1 }
				if = {
					limit = { check_variable = { MAN_lv_gongchangling < 2.5 } }
					add_resource = { type = steel amount = 30 }
				}
			}
			custom_effect_tooltip = MAN_tt_q_steel_10
			set_temp_variable = { MAN_t = 10 }
			multiply_temp_variable = { MAN_t = FROM.MAN_eff }
			multiply_temp_variable = { MAN_t = MAN_quota_mult }
			set_temp_variable = { MAN_t3 = MAN_qb_steel }
			clamp_temp_variable = { var = MAN_t3 min = 0 max = 0.5 }
			set_temp_variable = { MAN_t2 = MAN_t }
			multiply_temp_variable = { MAN_t2 = MAN_t3 }
			add_to_temp_variable = { MAN_t = MAN_t2 }
			add_to_variable = { MAN_q_steel = MAN_t }
			add_to_variable = { MAN_rail_built = 1 }
			custom_effect_tooltip = MAN_tt_spur
			add_to_variable = { MAN_lvc_gongchangling = 1 }
			custom_effect_tooltip = MAN_behemoth_risk_tt
			hidden_effect = {
				MAN_behemoth_accident_roll = yes
				MAN_plan_recalc = yes
				MAN_ui_refresh = yes
			}
		}
		ai_will_do = { base = 10 }
	}
