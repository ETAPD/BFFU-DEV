	# Always shown to eligible recipients while the US market is open, so the tab is reachable.
	#####################################

	USA_cnc_buyer_overview = {
		icon = USA_cnc_money
		allowed = { always = yes }
		visible = { USA = { has_country_flag = USA_cash_and_carry_active  has_war = no } }
		available = { custom_trigger_tooltip = { tooltip = USA_cnc_overview_hint_tt  always = no } }
		cost = 0
		fire_only_once = no

	#####################################
	# ====  RECIPIENT: BUY FROM THE US MARKET (per type)  ====
	# Eligible tags are governed by the USA_cnc_market_cat category's allowed block. All four
	# buys stay visible (greyed with reasons) so every recipient always has the four options.
	# On purchase the package leaves the market at once; equipment arrives after 90 days.
	#####################################

	USA_cnc_buy_infantry = {
		icon = USA_cnc_lend_lease
		allowed = { always = yes }

		visible = {
			USA = { has_country_flag = USA_cash_and_carry_active  has_war = no }
			NOT = { has_war_with = USA }
			NOT = { has_government = fascism }
		}

		available = {
			custom_trigger_tooltip = { tooltip = USA_cnc_buyer_at_war_tt  has_war = yes }
			custom_trigger_tooltip = {
				tooltip = USA_cnc_stock_inf_tt
				check_variable = { global.cnc_stock_inf > 0.5 }
			}
		}

		cost = 60
		days_re_enable = 120
		fire_only_once = no

		complete_effect = {
			subtract_from_variable = { global.cnc_stock_inf = 1 }
			country_event = { id = usa.9401 days = 90 }
			USA_cnc_purchase_cost = yes
		}

		ai_will_do = { factor = 4 modifier = { factor = 2 is_in_faction_with = USA } }
