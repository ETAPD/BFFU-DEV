	# Always shown to eligible recipients while the US market is open, so the tab is reachable.
	#####################################

	USA_cnc_buyer_overview = {
		icon = USA_cnc_money

	#####################################
	# ====  RECIPIENT: BUY FROM THE US MARKET (per type)  ====
	# Eligible tags are governed by the USA_cnc_market_cat category's allowed block.
	# Timed decision: 5 civilian factories in use for 90 days (civilian_factory_use); the
	# equipment is delivered by remove_effect when that ends. days_re_enable adds 30 more days.
	#####################################

	USA_cnc_buy_infantry = {
		icon = USA_cnc_lend_lease
		allowed = { always = yes }

		visible = {
			USA = { has_country_flag = USA_cash_and_carry_active  has_war = no }
			NOT = { has_war_with = USA }
			NOT = { has_government = fascism }
		}

		available = {
			custom_trigger_tooltip = {
				tooltip = USA_cnc_stock_inf_tt
				check_variable = { global.cnc_stock_inf > 0.5 }
			}
		}

		cost = 60
		days_remove = 90
		days_re_enable = 30
		fire_only_once = no

		modifier = {
			civilian_factory_use = 5
		}

		complete_effect = {
			# On purchase: remove the package from the market and grow the US war economy.
			subtract_from_variable = { global.cnc_stock_inf = 1 }
			USA_cnc_purchase_reward = yes
		}

		remove_effect = {
			# After 90 days: the equipment arrives.
			add_equipment_to_stockpile = { type = infantry_equipment amount = 6000 }
			add_equipment_to_stockpile = { type = squad_equipment    amount = 2000 }
		}

		ai_will_do = { factor = 0 } # player-only: the AI must not buy from the market
	}
