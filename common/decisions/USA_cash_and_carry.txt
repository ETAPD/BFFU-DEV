	#####################################
	# ====  MEMBERS: BUY FROM THE US MARKET (per type, paid from the pool)  ====
	# Eligible tags are governed by the USA_cnc_market_cat category's allowed block.
	# Timed decision: 5 civilian factories in use for 90 days (civilian_factory_use); the
	# lots bought are stored per type and delivered by remove_effect when that ends.
	# days_re_enable adds 30 more days.
	#####################################

	USA_cnc_buy_infantry = {
		icon = USA_cnc_lend_lease
		allowed = { always = yes }
		visible = {
			USA = { has_country_flag = USA_cash_and_carry_active  has_war = no }
			NOT = { has_war_with = USA }
			NOT = { has_government = fascism }
		}

		available = {
			custom_trigger_tooltip = {
				tooltip = USA_cnc_size_chosen_tt
				check_variable = { cnc_pkg_lots > 0.5 }
			}
			custom_trigger_tooltip = {
				tooltip = USA_cnc_stock_inf_tt
				NOT = { check_variable = { global.cnc_stock_inf < cnc_pkg_lots } }
			}
			# custom_cost_trigger only colours the cost; this is what actually blocks the purchase
			custom_trigger_tooltip = {
				tooltip = USA_cnc_pool_enough_tt
				NOT = { check_variable = { global.cnc_pp_pool < cnc_pkg_cost } }
			}
		}

		custom_cost_trigger = {
			NOT = { check_variable = { global.cnc_pp_pool < cnc_pkg_cost } }
		}
		custom_cost_text = USA_cnc_pool_cost

		days_remove = 90
		days_re_enable = 30
		fire_only_once = no

		modifier = {
			civilian_factory_use = 5
		}

		complete_effect = {
			# On purchase: pay from the pool and take the lots off the market.
			USA_cnc_pay_from_pool = yes
			set_variable = { cnc_lots_inf = cnc_pkg_lots }
			subtract_from_variable = { global.cnc_stock_inf = cnc_pkg_lots }
		}

		remove_effect = {
			# After 90 days: one package per lot bought arrives.
			custom_effect_tooltip = USA_cnc_deliver_inf_tt
			hidden_effect = {
				set_temp_variable = { cnc_deliver = cnc_lots_inf }
				while_loop_effect = {
					limit = { check_variable = { cnc_deliver > 0.5 } }
					add_equipment_to_stockpile = { type = infantry_equipment amount = 6000 }
					add_equipment_to_stockpile = { type = squad_equipment amount = 2000 }
					subtract_from_temp_variable = { cnc_deliver = 1 }
				}
			}
		}

		ai_will_do = { factor = 0 } # player-only: the AI must not buy from the market
	}
