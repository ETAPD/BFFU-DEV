				random_owned_controlled_state = {
					limit = { is_core_of = ROOT }
					add_extra_state_shared_building_slots = 1
					add_building_construction = { type = industrial_complex level = 1 instant_build = yes }
				}
				random_owned_controlled_state = {
					limit = { is_core_of = ROOT }
					add_extra_state_shared_building_slots = 1
					add_building_construction = { type = industrial_complex level = 1 instant_build = yes }

					limit = { is_core_of = ROOT }

					add_building_construction = { type = industrial_complex level = 1 instant_build = yes }
				}
				custom_effect_tooltip = USA_kaiser_contract_civ_3_tt
			}
			else_if = {
				limit = { check_variable = { USA_kaiser_trust > 1 } }
				add_ideas = USA_kaiser_contract_2
				set_country_flag = USA_kaiser_contract_tier_2
				# 2 civs in random states
				random_owned_controlled_state = {
					limit = { is_core_of = ROOT }
					add_extra_state_shared_building_slots = 1
					add_building_construction = { type = industrial_complex level = 1 instant_build = yes }
				}
				random_owned_controlled_state = {
					limit = { is_core_of = ROOT }
					add_extra_state_shared_building_slots = 1

				}
				custom_effect_tooltip = USA_kaiser_contract_civ_2_tt
			}
			else = {
				add_ideas = USA_kaiser_contract_1
				set_country_flag = USA_kaiser_contract_tier_1
				# 1 civ in random state
				random_owned_controlled_state = {
					limit = { is_core_of = ROOT }
					add_extra_state_shared_building_slots = 1
					add_building_construction = { type = industrial_complex level = 1 instant_build = yes }
				}
				custom_effect_tooltip = USA_kaiser_contract_civ_1_tt
			}
			add_to_variable = { var = USA_active_contracts value = 1 }
			set_country_flag = USA_kaiser_contract_active
		}

		remove_effect = {
			if = {
				limit = { has_country_flag = USA_kaiser_contract_tier_1 }
				remove_ideas = USA_kaiser_contract_1
				clr_country_flag = USA_kaiser_contract_tier_1
			}
			if = {
				limit = { has_country_flag = USA_kaiser_contract_tier_2 }
				remove_ideas = USA_kaiser_contract_2
				clr_country_flag = USA_kaiser_contract_tier_2
			}
			if = {
				limit = { has_country_flag = USA_kaiser_contract_tier_3 }
				remove_ideas = USA_kaiser_contract_3
				clr_country_flag = USA_kaiser_contract_tier_3
			}

			clr_country_flag = USA_kaiser_contract_active

				country_event = { id = usa_corp.1 }
			}
