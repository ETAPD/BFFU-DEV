	SIA_war_fervor_coup = {

		icon = generic_civil_support
		available = {
			hidden_trigger = {
				always = no
			}
		}
		priority = 10
		days_mission_timeout = 125

		activation = {
			has_war_support > 0.95 
			has_civil_war = no
			has_war = no
			has_idea = idea_SIA_war_fervor
		}

		cancel_trigger = {
			OR = {
				has_war_support < 0.95
				has_war = yes
				NOT = { has_idea = idea_SIA_war_fervor }
			}
		}

		fire_only_once = no

		is_good = no

		timeout_effect = {
			if = {
				limit = { has_idea = idea_SIA_war_fervor }
				add_war_support = -0.5
				if = { #The AI should avoid this at all costs on historical, but if for some reason it fails due to lack off pp safeguard on historical and punish in other ways
					limit = {
						is_ai = yes
						is_historical_focus_on = yes
					}
					add_ideas = volunteer_only
					add_ideas = civilian_economy
				}
				else = {
					SIA_start_opposing_civil_war = yes
				}

				remove_ideas = idea_SIA_war_fervor
			}
		}
	}

	SIA_urge_restraint = {

SIA_limited_war_cat = {
	SIA_start_limited_war_indochina = {

	    target_root_trigger = {
	       original_tag = SIA
	    }
	    visible = {
	    	has_global_flag = neutralized_focus_SIA_recover_indochina
	    }
	    target_array = ROOT.indochina_controllers
	    target_trigger = {
    		NOT = {
    			is_in_array = {
    				array = previous_limited_war_targets value = FROM
    			}
    		}

				is_controlled_by = FROM

	    	NOT = {
	    		is_ally_with = FROM
    			has_war_with = FROM
    			has_non_aggression_pact_with = FROM
    			FROM = { is_guaranteed_by = ROOT }
	    	}

    		custom_trigger_tooltip = {
    			tooltip = is_not_engaged_in_a_limited_war
	    		NOT = { has_variable = limited_war_target } 
    		}
	    }

	    highlight_states = {
	        highlight_states_trigger = {
		    	is_in_array = {
		    		array = ROOT.indochina_states
		    		value = THIS
		    	}
		    	is_controlled_by = FROM

	        }
	    }

	    cost = 20
	    complete_effect = {
	    	custom_effect_tooltip = SIA_start_limited_war_indochina_tt_main

	    	if = {
	    		limit = {
	    			is_in_faction = yes
	    			is_faction_leader = no
	    		}

	    		faction_leader = { set_variable = { ROOT.pre_limited_war_faction_leader = THIS } }
	    		leave_faction = yes
	    		custom_effect_tooltip = SIA_start_limited_war_rejoin_faction
	    		custom_effect_tooltip = newline_tt

	    	set_rule = { can_join_factions = no }

	    	custom_effect_tooltip = SIA_start_limited_war_can_join_faction
	    	#General cleanup
	    	clear_array = limited_war_states
	    	clear_variable = limited_war_target 
	    	clear_array = limited_war_target_allies

	    	#Save all states that is included in the limited war
	    	for_each_scope_loop = {
	    		array = indochina_states
	    		if = {
	    			limit = { is_controlled_by = FROM }
	
	    			add_to_array = { array = ROOT.limited_war_states value = THIS }
	    		}
	    	}

	    	set_variable = { limited_war_target = FROM } #is_target_of_limited_war

	    	if = { #overlord will always be called if target is a subject
	    		limit = {
	    			var:limited_war_target = { is_subject = yes }
	    		}

	    		var:limited_war_target = {
	    			overlord = {
	    				add_to_array = { array = ROOT.limited_war_target_allies value = THIS }
	    			}
	    		}
	    	}

	    	hidden_effect = {
	    		var:limited_war_target = {
	    			if = {
	    				limit = {
	    					is_in_faction = yes
	    				}

	    				every_allied_country = {
	    					
	    					limit = {
	    						NOT = {
	    							is_in_array = {
	    								array = ROOT.limited_war_target_allies
	    								value = THIS
	    							}
	    						}
	    					}
	    					add_to_array = { array = ROOT.limited_war_target_allies value = THIS }
	    				}
	    			}	
	    		}
	    	}

	    	
    		every_country = {
    			limit = { has_guaranteed = FROM }

    			FROM = { clear_array = countries_that_guaranteed_me_pre_limited_war }
    			add_to_array = {
    				array = FROM.countries_that_guaranteed_me_pre_limited_war
    				value = THIS
    			}
    			diplomatic_relation = { 
    				country = FROM
    				relation = guarantee 	
    				active = no 
    			}
    			

    		}
	    	
	    	declare_war_on = {
	    		target = FROM
	    		type = annex_everything
	    	}

	    	activate_mission = SIA_limited_war_indochina
	    	add_to_array = { previous_limited_war_targets = FROM }
	    	FROM = {
	    		if = {
	    			limit = { is_in_faction = no }
	    			if = {
	    				limit = { has_rule = can_join_factions }
	    				set_country_flag = SIA_limited_war_could_join_factions
	    			}
	    			set_rule = { can_join_factions = no }
	    		}
	    	}

	    }

	    ai_will_do = {
	    	base = 0
	    	modifier = {
	    		is_historical_focus_on = yes
	    		date > 1940.10.28

	    		add = 500
	    	}

	    	modifier = {
	    		VIC = { exists = yes }
	    		any_of_scopes = {
	    			array = indochina_controllers 
	    			tag = VIC
	    		}
	    		FROM = {
	    			tag = VIC
	    			is_in_faction = no
	    		}

	    		add = 100
	    	}

	    	modifier = {
	    		fighting_army_strength_ratio = {
	    			tag = FROM
	    			ratio > 0.3
	    		}

	    		add = 20
	    	}

	    	modifier = {
	    		fighting_army_strength_ratio = {
	    			tag = FROM
	    			ratio > 0.6
	    		}

	    		add = 30
	    	}

	    	modifier = {
	    		fighting_army_strength_ratio = {
	    			tag = FROM
	    			ratio > 0.9
	    		}

	    		add = 70
	    	}

	    	modifier = {
	    		FROM = {
	    			original_tag = JAP
	    		}
	    		factor = 0
	    	}
	    }
    }
