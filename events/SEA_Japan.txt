			}

			#	is_historical_focus_on = yes 
			#	factor = 0
			#}
		}

		FROM = {
			country_event = {
				# FROM refuses to fold
				id = SEA_japan_foreign_policy.6
				hours = 12
				random_hours = 24
			}
		}

	}
