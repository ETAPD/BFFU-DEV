add_namespace = rcc

# 29 July 1937: the Peace Preservation Corps at Tongzhou turns on its Japanese advisers.
country_event = {
	id = rcc.1
	title = rcc.1.t
	desc = rcc.1.d
	picture = GFX_report_event_BFFU_china_state_609
	is_triggered_only = yes
	option = {
		name = rcc.1.a
		set_country_flag = RCC_tongzhou_mutiny
		add_timed_idea = { idea = RCC_tongzhou_aftermath days = 180 }
		add_country_leader_role = {
			character = RCC_chi_zongmo
			promote_leader = yes
			country_leader = {
				ideology = despotism
				traits = { RCC_caretaker_of_tangshan }
				expire = "1965.1.1.1"
			}
		}
	}
}
