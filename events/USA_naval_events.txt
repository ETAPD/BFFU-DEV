###########################
# USA Naval Events
###########################

add_namespace = USA_naval

# ── USA_naval.1 ─────────────────────────────────────────────────────────────
# "The Future of the United States Navy"
# Fires 3 days after Congress Passes the Bureau of Ships completes.
# Three mutually-exclusive choices autocomplete one of the three child focuses.
country_event = {
	id = USA_naval.1
	title = USA_naval.1.t
	desc = USA_naval.1.d
	picture = GFX_report_event_naval_treaty

	is_triggered_only = yes

	option = {				# Build for FY40
		name = USA_naval.1.a
		ai_chance = { factor = 30 }
		complete_national_focus = USA_build_for_fy40
	}

	option = {				# A Middle Ground
		name = USA_naval.1.b
		ai_chance = { factor = 40 }
		complete_national_focus = USA_a_middle_ground
	}

	option = {				# Plan for FY42
		name = USA_naval.1.c
		ai_chance = { factor = 30 }
		complete_national_focus = USA_plan_for_fy42
	}
}

# ── USA_naval.2 ─────────────────────────────────────────────────────────────
# Fiscal 40 comes to a close: checked monthly. From April 1940 the Fiscal Year 40 Funding
# spirit ends and the funds are reprioritised into War Economy of the United States.
country_event = {
	id = USA_naval.2
	title = USA_naval.2.t
	desc = USA_naval.2.d
	picture = GFX_report_event_naval_treaty
	fire_only_once = yes

	trigger = {
		tag = USA
		has_idea = USA_fy40_funding
		date > 1940.3.31
	}

	is_triggered_only = yes

	option = {
		name = USA_naval.2.a
		remove_ideas = USA_fy40_funding
		custom_effect_tooltip = USA_modify_war_economy_tt
		add_to_variable = { USA_we_output = 0.05 tooltip = industrial_capacity_factory_tt }
		add_to_variable = { USA_we_research = 0.15 tooltip = research_speed_factor_tt }
		hidden_effect = { force_update_dynamic_modifier = yes }
	}
}
