	desc = bulgaria.4.d

	trigger = {
		tag = BUL
		NOT = { has_country_flag = bulgaria.4_fired }
		BUL = {
			NOT = { is_in_faction_with = ROM }
		}
		BUL = {
			NOT = { is_subject_of = ROM }
		}
		ROM = {
			OR = { 
				has_country_flag = second_vienna_accepted_north

	}	
	#dont make demands
	option = {
		ai_chance = { factor = 0 }
		name = bulgaria.4.b
	}
}
