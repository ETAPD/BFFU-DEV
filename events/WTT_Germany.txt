	desc = wtt_germany.63.desc
	picture = GFX_report_event_generic_peaceful_annexation

	is_triggered_only = yes
	
	option = {
		name = wtt_germany.63.a
		IF = {
			limit = {
				GER = {
					has_completed_focus = GER_reintegrate_luxemburg_and_alsace_lorraine
				}
			}
			effect_tooltip = {
				GER = {
					add_state_core = 8
					add_state_core = 28
				}
			}
		}
		ELSE = {
			effect_tooltip = {
				GER = {
					add_state_core = 28
				}
			}
		}
	}
}

	id = wtt_germany.1001
	title = wtt_germany.1001.t
	desc = wtt_germany.1001.desc
	picture = GFX_report_event_hindenburg_flames

	is_triggered_only = yes

	# The Kaiser is dead... Long live the Kaiserin!
	option = {
		name = wtt_germany.1001.a

		add_stability = -0.1
		kill_country_leader = yes

		GER_victoria = {
			add_country_leader_role = {
				country_leader = {
					expire = "1965.1.1"

						kaiserin_of_the_people
						victoria_iii # Third Victoria counting from Queen Victoria hurr durr

				promote_leader = yes
			}
		}

		mark_focus_tree_layout_dirty = yes

		hidden_effect = { set_global_flag = hindenburg_destroyed }

	}
}
