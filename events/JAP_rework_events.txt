# ===== The Melancholy Prince - Konoe's second-term decline (JAP_rework.20-.24) =====
# Fired from JAP_the_prince_returns. Konoe begins on tier 1 and decays: tier 2 after
# 140 days, tier 3 after a further 189 days, tier 4 gated to 22 July 1940 (the hidden
# poller JAP_rework.24, scheduled at startup, holds tier 4 until both that date has
# passed and tier 3 has been reached). All stages stop if Konoe is no longer leader.

# Tier 1 - The prince returns in renewed vigour

	id = JAP_rework.20
	title = JAP_rework.20.t
	desc = JAP_rework.20.desc
	picture = GFX_report_event_japanese_politicians

	is_triggered_only = yes
	fire_only_once = yes

	trigger = {
		has_country_leader = { character = JAP_fumimaro_konoe ruling_only = yes }
	}

	immediate = {
		remove_country_leader_trait = JAP_ambitious_political_views
		add_country_leader_trait = JAP_melancholy_prince_1
		country_event = { id = JAP_rework.21 days = 140 }
	}

	option = {
		name = JAP_rework.20.a
		ai_chance = { base = 1 }
	}
}

# Hidden date-gate for tier 4. Scheduled at startup for 22 July 1940 (day 1664).
# Fires tier 4 if Konoe rules and has reached tier 3; if tier 3 has not yet been
# reached on that date it re-checks every few days until it has.
country_event = {
	id = JAP_rework.24
	hidden = yes
	is_triggered_only = yes

	immediate = {
		if = {
			limit = {
				has_country_leader = { character = JAP_fumimaro_konoe ruling_only = yes }
				has_country_leader_with_trait = JAP_melancholy_prince_3
			}
			country_event = { id = JAP_rework.23 hours = 4 }
		}
		else_if = {
			limit = {
				has_country_leader = { character = JAP_fumimaro_konoe ruling_only = yes }
				OR = {
					has_country_leader_with_trait = JAP_melancholy_prince_1
					has_country_leader_with_trait = JAP_melancholy_prince_2
				}
			}
			country_event = { id = JAP_rework.24 days = 3 }
		}
	}
}

# Japan opens the war with China - 29 December 1937 (Second Sino-Japanese War).
# Scheduled at startup (JAP_rework_on_actions). The declaration trips on_declare_war
# (test_on_actions) which plays the speech/song and applies the Surprise Invasion idea.
country_event = {
	id = JAP_rework.30
	hidden = yes
	is_triggered_only = yes
	fire_only_once = yes

	trigger = {
		tag = JAP
		CHI = { exists = yes }
		NOT = { has_war_with = CHI }
	}

	immediate = {
		declare_war_on = { target = CHI type = annex_everything }
		# The China-war army strain (-65% XP / -40% division XP) and the Kwantung veto are applied by the
		# on_declare_war handler (test_on_actions), so they fire for ANY JAP->CHI declaration - this scheduled
		# fallback declare or the historical japan.1 Marco Polo Bridge chain.
	}
}
